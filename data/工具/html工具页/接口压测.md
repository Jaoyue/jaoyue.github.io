        // Web Worker管理器
        class WorkerManager {
            constructor(workerCount) {
                this.workerCount = workerCount;
                this.workers = [];
                this.requestId = 0;
                this.callbacks = new Map();
                this.workerStates = Array(workerCount).fill('idle');
            }
            
            // 初始化Worker
            async init() {
                for (let i = 0; i < this.workerCount; i++) {
                    const worker = new Worker(URL.createObjectURL(new Blob([`
                        self.onmessage = async function(e) {
                            const { id, url, method, headers, body } = e.data;
                            const startTime = Date.now();
                            
                            try {
                                const requestOptions = {
                                    method: method || 'GET',
                                    headers: headers || {},
                                    mode: 'cors',
                                    cache: 'no-cache'
                                };
                                
                                // 如果是POST/PUT/PATCH请求且有body，添加body
                                if (['POST', 'PUT', 'PATCH'].includes(method) && body) {
                                    requestOptions.body = JSON.stringify(body);
                                }
                                
                                const response = await fetch(url, requestOptions);
                                
                                const endTime = Date.now();
                                const duration = endTime - startTime;
                                
                                // 检查响应状态
                                if (response.ok) {
                                    postMessage({
                                        id,
                                        success: true,
                                        duration,
                                        status: response.status
                                    });
                                } else {
                                    postMessage({
                                        id,
                                        success: false,
                                        duration,
                                        status: response.status,
                                        error: \`HTTP错误: \${response.status}\`
                                    });
                                }
                            } catch (error) {
                                const endTime = Date.now();
                                postMessage({
                                    id,
                                    success: false,
                                    duration: endTime - startTime,
                                    error: error.message
                                });
                            }
                        };
                    `], { type: 'application/javascript' })));
                    
                    worker.onmessage = (e) => {
                        const { id, success, duration, error } = e.data;
                        if (this.callbacks.has(id)) {
                            const { resolve, reject, workerIndex } = this.callbacks.get(id);
                            this.callbacks.delete(id);
                            
                            // 更新Worker状态
                            this.workerStates[workerIndex] = 'idle';
                            this.updateWorkerStatus();
                            
                            if (success) {
                                resolve({ duration });
                            } else {
                                reject(new Error(error || '请求失败'));
                            }
                        }
                    };
                    
                    this.workers.push(worker);
                }
                
                this.updateWorkerStatus();
            }
            
            // 发送请求
            sendRequest(url, method = 'GET', headers = {}, body = null) {
                return new Promise((resolve, reject) => {
                    const id = this.requestId++;
                    
                    // 轮询分配请求给Worker
                    const workerIndex = id % this.workerCount;
                    this.workerStates[workerIndex] = 'busy';
                    this.updateWorkerStatus();
                    
                    this.callbacks.set(id, { resolve, reject, workerIndex });
                    this.workers[workerIndex].postMessage({ id, url, method, headers, body });
                });
            }
            
            // 更新Worker状态显示
            updateWorkerStatus() {
                workerStatusEl.innerHTML = '';
                
                this.workerStates.forEach((state, index) => {
                    const workerEl = document.createElement('div');
                    workerEl.className = `worker ${state === 'busy' ? 'active' : ''}`;
                    workerEl.innerHTML = `
                        Worker ${index + 1}: 
                        <span class="${state === 'busy' ? 'worker-busy' : 'worker-idle'}">
                            ${state === 'busy' ? '忙碌' : '空闲'}
                        </span>
                    `;
                    workerStatusEl.appendChild(workerEl);
                });
            }
            
            // 终止所有Worker
            terminate() {
                this.workers.forEach(worker => worker.terminate());
                this.workers = [];
                this.callbacks.clear();
            }
        }

                    body = JSON.parse(bodyText);
                } catch (e) {
                    addLog(`JSON解析错误: ${e.message}`, 'failed');
                    isTesting = false;
                    startBtn.disabled = false;
                    stopBtn.disabled = true;
                    return;
                }
            }
            
            totalEl.textContent = totalRequests;
            
            // 初始化Worker管理器
            workerManager = new WorkerManager(workerCount);
            await workerManager.init();
            
            addLog(`已创建 ${workerCount} 个Web Worker，每个Worker有6个连接限制`, 'info');
            addLog(`总并发限制: ${workerCount * 6} 个连接`, 'info');
            addLog(`请求方法: ${method}`, 'info');
            addLog(`Header数量: ${Object.keys(headers).length}`, 'info');
            
            // 执行测试
            const promises = [];
            
            for (let i = 0; i < totalRequests; i++) {
                const requestNumber = i + 1;
                const url = `${apiUrl}?request=${requestNumber}&_=${Date.now()}`;
                
                const promise = workerManager.sendRequest(url, method, headers, body)
                    .then(result => {
                        completedRequests++;
                        successCount++;
                        totalDuration += result.duration;
                        updateProgress();
                        addLog(`请求 ${requestNumber} 成功, 耗时: ${result.duration}ms`, 'success');
                    })
                    .catch(error => {
                        completedRequests++;
                        failedCount++;
                        updateProgress();
                        addLog(`请求 ${requestNumber} 失败: ${error.message}`, 'failed');
                    });
                
                promises.push(promise);
                
                // 控制并发数
                if (promises.length >= concurrentLimit) {
                    await Promise.allSettled(promises.splice(0, concurrentLimit));
                }
                
                // 检查是否停止
                if (!isTesting) break;
            }
            
            // 等待剩余请求完成
            await Promise.allSettled(promises);
            
            // 测试完成
            const endTime = Date.now();
            const totalTime = endTime - startTime;
            
            addLog(`测试完成! 总耗时: ${totalTime}ms, 成功: ${successCount}, 失败: ${failedCount}`, 'info');
            
            isTesting = false;
            startBtn.disabled = false;
            stopBtn.disabled = true;
            
            // 终止Worker
            if (workerManager) {
                workerManager.terminate();
                workerManager = null;
            }
        }
        
        // 停止测试
        function stopTest() {
            if (!isTesting) return;
            
            isTesting = false;
            addLog('测试已停止', 'info');
            
            if (workerManager) {
                workerManager.terminate();
                workerManager = null;
            }
            
            startBtn.disabled = false;
            stopBtn.disabled = true;
        }
        
        // 初始化
        function init() {
            // 事件监听
            startBtn.addEventListener('click', startTest);
            stopBtn.addEventListener('click', stopTest);
            validateJsonBtn.addEventListener('click', validateJson);
            
            // 监听配置变化，更新预览
            document.getElementById('requestMethod').addEventListener('change', updateRequestPreview);
            document.getElementById('apiUrl').addEventListener('input', updateRequestPreview);
            requestBody.addEventListener('input', updateRequestPreview);
            
            // 初始化Header管理和标签页
            initHeaderManagement();
            initTabs();
            
            // 更新进度
            updateProgress();
        }
        
        // 页面加载完成后初始化
        document.addEventListener('DOMContentLoaded', init);
    </script>
