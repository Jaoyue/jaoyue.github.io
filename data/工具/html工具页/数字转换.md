        function selectMode(mode) {
            // 更新选中状态
            document.querySelectorAll('.mode-option').forEach(option => {
                option.classList.remove('active');
            });
            event.currentTarget.classList.add('active');
            
            // 选中对应的单选按钮
            const radioButton = event.currentTarget.querySelector('input[type="radio"]');
            if (radioButton) {
                radioButton.checked = true;
            }
            
            // 更新示例内容
            updateExample(mode);
        }

        function updateExample(mode) {
            const exampleContent = document.getElementById('exampleContent');
            let inputExample, outputExample;

            switch(mode) {
                case 'single':
                    inputExample = `121710079438079837008<br>121710143163914208008<br>121710145110502771001<br>121727158130166001007`;
                    outputExample = `'121710079438079837008',<br>'121710143163914208008',<br>'121710145110502771001',<br>'121727158130166001007'`;
                    break;
                case 'double':
                    inputExample = `121710079438079837008<br>121710143163914208008<br>121710145110502771001<br>121727158130166001007`;
                    outputExample = `"121710079438079837008",<br>"121710143163914208008",<br>"121710145110502771001",<br>"121727158130166001007"`;
                    break;
                case 'singleToDouble':
                    inputExample = `'121710079438079837008', '121710143163914208008', '121710145110502771001', '121727158130166001007'`;
                    outputExample = `"121710079438079837008",<br>"121710143163914208008",<br>"121710145110502771001",<br>"121727158130166001007"`;
                    break;
                case 'commaSeparated':
                    inputExample = `121710079438079837008<br>121710143163914208008<br>121710145110502771001<br>121727158130166001007`;
                    outputExample = `121710079438079837008,121710143163914208008,121710145110502771001,121727158130166001007`;
                    break;
                case 'normalize':
                    inputExample = `"121710079438079837008"<br>"121710143163914208008"<br>”121710145110502771001“"<br>"121727158130166001007"`;
                    outputExample = `121710079438079837008<br>121710143163914208008<br>121710145110502771001<br>121727158130166001007`;
                    break;
            }

            exampleContent.innerHTML = `
                <div class="example-input">
                    <strong>输入格式：</strong><br>
                    ${inputExample}
                </div>
                <div class="example-output">
                    <strong>输出格式：</strong><br>
                    ${outputExample}
                </div>
            `;
        }

        function convertNumbers() {
            const input = document.getElementById('inputNumbers').value;
            const output = document.getElementById('outputNumbers');
            const selectedMode = document.querySelector('input[name="convertMode"]:checked').value;
            
            if (!input.trim()) {
                alert('请先输入要转换的内容！');
                return;
            }
            
            let result;
            const lines = input.split('\n')
                .map(line => line.trim())
                .filter(line => line !== '');

            switch(selectedMode) {
                case 'single':
                    result = lines.map(line => `'${line}'`).join(',\n');
                    break;
                case 'double':
                    result = lines.map(line => `"${line}"`).join(',\n');
                    break;
                case 'singleToDouble':
                    const allText = lines.join(' ');
                    const matches = allText.match(/'[^']*'/g);
                    if (matches) {
                        result = matches.map(match => {
                            const content = match.slice(1, -1);
                            return `"${content}"`;
                        }).join(',\n');
                    } else {
                        result = lines.map(line => {
                            const cleanLine = line.replace(/^'|'$/g, '').replace(/,$/, '');
                            return `"${cleanLine}"`;
                        }).join(',\n');
                    }
                    break;
                case 'commaSeparated':
                    result = lines.join(',');
                    break;
                case 'normalize':
                    result = lines.map(line => line.replace(/['"“”]/g, '').trim()).join('\n');
                    break;
                default:
                    result = lines.map(line => `'${line}'`).join(',\n');
            }
            
            output.value = result;
            
            // 添加复制到剪贴板的功能
            output.select();
            output.setSelectionRange(0, 99999); // 对移动设备
            
            try {
                document.execCommand('copy');
                showNotification('转换完成！结果已复制到剪贴板');
            } catch (err) {
                showNotification('转换完成！请手动复制结果');
            }
        }
        
        function showNotification(message) {
            // 创建通知元素
            const notification = document.createElement('div');
            notification.textContent = message;
            notification.style.cssText = `
                position: fixed;
                top: 20px;
                right: 20px;
                background: #4facfe;
                color: white;
                padding: 15px 25px;
                border-radius: 8px;
                box-shadow: 0 5px 15px rgba(0,0,0,0.2);
                z-index: 1000;
                font-weight: 600;
                animation: slideIn 0.3s ease;
            `;
            
            // 添加动画样式
            const style = document.createElement('style');
            style.textContent = `
                @keyframes slideIn {
                    from { transform: translateX(100%); opacity: 0; }
                    to { transform: translateX(0); opacity: 1; }
                }
            `;
            document.head.appendChild(style);
            
            document.body.appendChild(notification);
            
            // 3秒后移除通知
            setTimeout(() => {
                notification.remove();
                style.remove();
            }, 3000);
        }

        document.addEventListener('DOMContentLoaded', function() {
