    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const fetchBtn = document.getElementById('fetchBtn');
            const jsonInput = document.getElementById('jsonInput');
            const filesContainer = document.getElementById('filesContainer');
            const fileCount = document.getElementById('fileCount');
            const statusDiv = document.getElementById('status');
            const progressContainer = document.getElementById('progressContainer');
            const progressBar = document.getElementById('progressBar');

            // 点击“获取文件列表”按钮时触发
            fetchBtn.addEventListener('click', function() {
                filesContainer.innerHTML = ''; // 清空原有列表
                statusDiv.textContent = '';
                progressContainer.style.display = 'none';

                try {
                    const inputText = jsonInput.value.trim();
                    if (!inputText) {
                        statusDiv.textContent = '请输入有效的JSON数组';
                        return;
                    }
                    // 解析JSON
                    const files = JSON.parse(inputText);
                    // 验证是否为数组
                    if (!Array.isArray(files)) {
                        throw new Error('输入的JSON必须是一个数组');
                    }
                    // 验证每个文件对象的结构（必须包含name和url）
                    files.forEach(file => {
                        if (!file.name || !file.url) {
                            throw new Error('每个文件对象必须包含name和url字段');
                        }
                    });

                    // 生成文件列表
                    fileCount.textContent = files.length;
                    files.forEach((file, index) => {
                        const fileItem = document.createElement('div');
                        fileItem.className = 'file-item';

                        const fileNameSpan = document.createElement('span');
                        fileNameSpan.className = 'file-name';
                        fileNameSpan.textContent = `${index + 1}. ${file.name}`;

                        const downloadBtn = document.createElement('button');
                        downloadBtn.className = 'download-btn';
                        downloadBtn.textContent = '下载';
                        downloadBtn.dataset.url = file.url;
                        downloadBtn.dataset.filename = file.name;

                        fileItem.appendChild(fileNameSpan);
                        fileItem.appendChild(downloadBtn);
                        filesContainer.appendChild(fileItem);
                    });

                    statusDiv.textContent = `成功加载 ${files.length} 个文件`;
                } catch (error) {
                    statusDiv.textContent = `解析失败: ${error.message}`;
                }
            });

            // 下载文件的核心逻辑
            async function downloadFile(url, filename) {
                statusDiv.textContent = `正在下载: ${filename}`;
                progressContainer.style.display = 'block';
                progressBar.style.width = '0%';

                try {
                    const response = await fetch(url, {
                        method: 'GET',
                        headers: { 'Content-Type': 'application/octet-stream' }
                    });

                    if (!response.ok) {
                        throw new Error(`HTTP错误: ${response.status}`);
                    }

                    const contentLength = response.headers.get('content-length');
                    let loaded = 0;

                    // 流式下载并更新进度
                    const reader = response.body.getReader();
                    const chunks = [];
                    while (true) {
                        const { done, value } = await reader.read();
                        if (done) break;
                        chunks.push(value);
                        loaded += value.length;
                        if (contentLength) {
                            const progress = Math.round((loaded / contentLength) * 100);
                            progressBar.style.width = `${progress}%`;
                        }
                    }

                    // 生成Blob并触发下载
                    const blob = new Blob(chunks);
                    const downloadUrl = URL.createObjectURL(blob);
                    const a = document.createElement('a');
                    a.href = downloadUrl;
                    a.download = filename;
                    document.body.appendChild(a);
                    a.click();
                    document.body.removeChild(a);
                    URL.revokeObjectURL(downloadUrl);

                    statusDiv.textContent = `下载完成: ${filename}`;
                    progressBar.style.width = '100%';

                    setTimeout(() => {
                        progressContainer.style.display = 'none';
                    }, 2000);

                    return true;
                } catch (error) {
                    console.error('下载失败:', error);
                    statusDiv.textContent = `下载失败: ${filename} - ${error.message}`;
                    progressBar.style.backgroundColor = '#ff3333';
                    return false;
                }
            }

            // 下载按钮的点击事件（事件委托，支持动态生成的按钮）
            document.addEventListener('click', function(e) {
                if (e.target.classList.contains('download-btn')) {
                    const btn = e.target;
                    const url = btn.dataset.url;
                    const filename = btn.dataset.filename;

                    btn.disabled = true;
                    btn.textContent = '下载中...';

                    downloadFile(url, filename).finally(() => {
                        setTimeout(() => {
                            btn.disabled = false;
                            btn.textContent = '下载';
                        }, 2000);
                    });
                }
            });
        });
    </script>
