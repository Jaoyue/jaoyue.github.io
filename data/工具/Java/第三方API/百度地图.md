@RequestMapping(value = "/searchMap", method = RequestMethod.GET)//匹配访问路径
    @ResponseBody//返回数据格式
    public ResultResponse searchMap(String query,
                                    String tag,
                                    String region,
                                    String output,
                                    String city_limit,
                                    String ak) {

        ResultResponse resultResponse = new ResultResponse();
        logger.info("v1.private.customer.searchMap,query:"+query+",tag:"+tag+",region:"+region+",output:"+output+",ak:"+ak+",city_limit:"+city_limit);
        try {
            Map<String,String> param = new HashMap<>();
            param.put("query",query);
            param.put("tag",tag);
            param.put("region",region);
            param.put("output",output);
            param.put("ak",ak);
            param.put("city_limit",city_limit);
            String url = ConfigCacheManager.getConfigValue(RuleConfigConstant.BAIDU_MAP_API_URL);
            HttpResult httpResult = HttpClientUtils.doGet(url, param);
            JSONObject jsonObject = JSONObject.parseObject(httpResult.getBody(), JSONObject.class);
            resultResponse.setData(jsonObject);
            resultResponse.setResult(HttpResultStatus.SUCCESS);
            return resultResponse;
        } catch (Exception e) {
            logger.error("v1.private.customer.searchMap error,query:"+query+",tag:"+tag+",region:"+region+",output:"+output+",ak:"+ak,e);
            resultResponse.setResult(HttpResultStatus.FAIL);
        }
        return resultResponse;
    }
