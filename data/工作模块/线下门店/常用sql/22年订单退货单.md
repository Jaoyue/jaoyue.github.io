```plsql
select 
o.order_no 订单号,
s.name 门店名称, 
(case 
when o.order_type = 'RETAIL_ORDER' then '零售订单'
when o.order_type = 'PURCHASE_ORDER' then '团购订单'
when o.order_type = 'COMMUNITY_ORDER' then '社群订单'
when o.order_type = 'CENTRALIZED_ORDER' then '集采订单'
when o.order_type = 'PRODUCT_ORDER' then '大单机订单'
when o.order_type = 'CITY_EXHIBITION_HALL' then '城市展厅'
when o.order_type = 'TRADE_ORDER' then '换货订单' end) 订单类型,
FROM_UNIXTIME(o.order_time/1000,'%Y-%m-%d %h:%i:%s') 创建时间,
(case 
when o.order_source = 'APP' then 'APP店员下单'
when o.order_source = 'IMPORT' then '导入订单'
when o.order_source = 'CITY_EXHIBITION_HALL' then '城市展厅' end) 订单来源,
c.name 公司名称,
(case 
when i.is_gift = 'Y' then '是'
when i.is_gift = 'N' then '否' end) 是否赠品,
(case 
when o.status = 'FINISHED' then '交易完成'
when o.status = 'WAITING_PAY' then '待付款'
when o.status = 'PAYING' then '支付中'
when o.status = 'PAY_FAIL' then '支付失败'
when o.status = 'CANCEL' then '作废'
when o.status = 'WAITING_PICK_UP' then '待自提' end) 订单状态,
o.customer_name 客户姓名,
o.customer_mobile 客户手机,
o.customer_district_id 客户所在区编号,
o.customer_address 客户地址,
i.material_no 物料号,
(case when i.product_classification = 'XCQ' then '吸尘器'
when i.product_classification = 'QXJ' then '洗地机'
when i.product_classification = 'CFJ' then '吹风机'
when i.product_classification = 'LLJ' then '料理机'
when i.product_classification = 'MYY' then '美眼仪'
when i.product_classification = 'MFS' then '美发梳'
when i.product_classification = 'OTHER' then '其他'
end) 商品分类,
i.product_name 商品名称,
i.sale_price_down 销售指导价_下价,
i.sale_price_up 销售指导价_上价,
i.price 单价,
i.purchase_price 进价,
i.count 数量,
(case 
when o.pay_type = 'CASH' then '现金'
when o.pay_type = 'GIRO' then '转账'
when o.pay_type = 'ALI_PAY' then '支付宝'
when o.pay_type = 'WX' then '微信' end) 支付方式,
i.total 支付金额,
FROM_UNIXTIME(o.pay_time/1000,'%Y-%m-%d %h:%i:%s') 支付时间,
(case 
when o.ship_type = 'EXPRESS' then '快递'
when o.ship_type = 'STORE_SELF' then '门店自提' end) 发货方式,
FROM_UNIXTIME(o.ship_time/1000,'%Y-%m-%d %h:%i:%s') 发货时间
from sp_sales_order_item i 
join sp_sales_order o on i.sales_order_no = o.order_no
join sp_store_info s on o.store_id = s.id
join sp_company c on o.company_no = c.sap_company_no
where o.order_time >= 1640966400000 and o.order_time< 1672502400000
```

```plsql
select 
o.order_no 退款单号,
o.sales_order_no 关联销售单号,
s.name 门店名称, 
c.name 所属公司,
FROM_UNIXTIME(o.create_time/1000,'%Y-%m-%d %h:%i:%s') 创建时间,
(case 
when o.status = 'FINISHED' then '已完成'
when o.status = 'WAITING' then '待确认'
when o.status = 'CANCEL' then '作废' end) 订单状态,
o.customer_name 客户姓名,
o.customer_mobile 客户手机,
o.customer_district_id 客户所在区编号,
o.customer_address 客户地址,
i.material_no 物料号,
(case when i.product_classification = 'XCQ' then '吸尘器'
when i.product_classification = 'QXJ' then '洗地机'
when i.product_classification = 'CFJ' then '吹风机'
when i.product_classification = 'LLJ' then '料理机'
when i.product_classification = 'MYY' then '美眼仪'
when i.product_classification = 'MFS' then '美发梳'
when i.product_classification = 'OTHER' then '其他'
end) 商品分类,
i.product_name 商品名称,
i.sale_price_down 销售指导价_下价,
i.sale_price_up 销售指导价_上价,
(case 
when i.is_gift = 'Y' then '是'
when i.is_gift = 'N' then '否' end) 是否赠品,
i.price 单价,
i.purchase_price 进价,
i.count 数量,
i.total 退款金额,
(case 
when o.refund_type = 'CASH' then '现金'
when o.refund_type = 'GIRO' then '转账' end) 退款金额类型,
o.reason 退货原因,
FROM_UNIXTIME(o.refund_time/1000,'%Y-%m-%d %h:%i:%s') 退款时间
from sp_refund_order_item i 
join sp_refund_order o on i.refund_order_no = o.order_no
join sp_store_info s on o.store_id = s.id
join sp_company c on o.company_no = c.sap_company_no
join sp_product p on i.material_no = p.material_no
where o.create_time >= 1640966400000 and o.create_time< 1672502400000



```
