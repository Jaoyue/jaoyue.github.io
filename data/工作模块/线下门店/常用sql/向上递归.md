### 传入子id, 查询它所有的父级数据
### mysql低版本中没有递归函数, 可以通过存储过程或者查询sql实现向上递归查询
```plsql

DROP FUNCTION IF EXISTS `getParentList`;  
DELIMITER $$  
  
CREATE FUNCTION `getParentList`(rootId VARCHAR(10000)) RETURNS VARCHAR(10000) CHARSET utf8  
    BEGIN   
        DECLARE sParentList VARCHAR(10000);   
        DECLARE sParentTemp VARCHAR(10000);   
          
        SET sParentTemp =rootId;   
          
        WHILE sParentTemp IS NOT NULL DO   
            IF (sParentList IS NOT NULL) THEN   
                SET sParentList = CONCAT(sParentTemp,',',sParentList);   
            ELSE   
                SET sParentList = CONCAT(sParentTemp);   
            END IF;   
              
            SELECT GROUP_CONCAT(superior_category_id) INTO sParentTemp FROM sp_train_question_classification WHERE FIND_IN_SET(id,sParentTemp)>0;   
        END WHILE;   
        RETURN sParentList;   
    END$$  
  
DELIMITER ;  
  
-- 调用  
select GROUP_CONCAT(t.category_name SEPARATOR '/') from(
SELECT * FROM sp_train_question_classification WHERE FIND_IN_SET(id, getParentList('31'))
) t group by t.category_status
 

```

```plsql

-- 查询sql
select GROUP_CONCAT(t.category_name SEPARATOR '/') from(
SELECT T2.*
FROM ( 
    SELECT 
        @r AS _id, 
        (SELECT @r := superior_category_id FROM sp_train_exam_classification WHERE id = _id) AS superior_category_id, 
        @l := @l + 1 AS lvl 
    FROM 
        (SELECT @r := '4', @l := 0) vars, 
        sp_train_exam_classification h 
    WHERE @r <> 0) T1 
JOIN sp_train_exam_classification T2 
ON T1._id = T2.id 
ORDER BY T1.lvl DESC ) t
group by t.category_status
```
