```plsql
select o.order_no 订单号,
FROM_UNIXTIME(o.order_time/1000,'%Y-%m-%d %h:%i:%s') 业务时间,
l.operator_name operator_name,
u.name 审核人,
FROM_UNIXTIME(o.audit_time/1000,'%Y-%m-%d %h:%i:%s') 审核时间
from sp_sales_order o
left join sp_sales_order_log l on o.id = l.order_id
and l.content = '审核订单' 
left join sp_biz_staff u on l.operator_name = u.user_name
where
 o.order_time >= 1656604800000
```
