```plsql
select s.order_no 订单号,
(case when s.status = 'FINISHED' then '交易完成' 
when s.status = 'WAITING_PAY' then '待付款'
when s.status = 'PAYING' then '支付中'
when s.status = 'PAY_FAIL' then '支付失败'
when s.status = 'CANCEL' then '作废'
when s.status = 'WAITING_PICK_UP' then '待自提' end
) 订单状态,st.store_no 门店编码,st.store_name 门店名称,
(case when st.sales_region = 'DONGBEI' then '沈阳分部'
when st.sales_region = 'NHUABEI' then '北京分部'
when st.sales_region = 'HUAZHONG' then '武汉分部'
when st.sales_region = 'HUDONGYI' then '济南分部'
when st.sales_region = 'HUDONGER' then '西安分部'
when st.sales_region = 'HUANAN' then '广州分部'
when st.sales_region = 'XINAN' then '成都分部'
when st.sales_region = 'XIBEI' then '太原分部'
when st.sales_region = 'SUZHOU' then '苏州分部'
when st.sales_region = 'CHINA' then '总部'
when st.sales_region = 'SUNINGDDJ' then '苏宁大单机'
when st.sales_region = 'KUNMING' then '昆明分部'
when st.sales_region = 'ZHENGZHOU' then '郑州分部'
when st.sales_region = 'NEWCHANNEL' then '新渠道'
when st.sales_region = 'HANZHOU' then '杭州分部'
when st.sales_region = 'NANNING' then '南宁分部'
when st.sales_region = 'CHANGSHA' then '长沙分部'
when st.sales_region = 'SHANGHAI' then '上海分部'
when st.sales_region = 'HEFEI' then '合肥分部'
when st.sales_region = 'NANJING' then '南京分部'
when st.sales_region = 'XIAMEN' then '厦门分部'
when st.sales_region = 'KA' then '大客户' end
) 所属大区,
(case when st.workspace_type = 'REGION' then '大区'
when st.workspace_type = 'LARGE_CLIENT' then '大客户'
when st.workspace_type = 'NEW_CHANNEL' then '新渠道' end
) 门店归属,
FROM_UNIXTIME(s.order_time/1000,'%Y-%m-%d %h:%i:%s') 业务时间,
st.company_name 所属公司,
(case when st.company_type = 'DIST' then '经销'
else '直营' end ) 公司类型,
(case when s.order_type = 'RETAIL_ORDER' then '零售订单'
when s.order_type = 'PURCHASE_ORDER' then '团购订单'
when s.order_type = 'COMMUNITY_ORDER' then '社群订单'
when s.order_type = 'CENTRALIZED_ORDER' then '集采订单'
when s.order_type = 'PRODUCT_ORDER' then '大单机订单'
when s.order_type = 'CITY_EXHIBITION_HALL' then '城市展厅'
when s.order_type = 'TRADE_ORDER' then '换货订单' end
) 订单类型,s.remark 订单备注,
customer_name 顾客姓名,customer_mobile 顾客手机,customer_address 顾客地址,product_count 主数量,i.material_no 物料号, 
(case when i.product_classification = 'XCQ' then '吸尘器'
when i.product_classification = 'QXJ' then '洗地机'
when i.product_classification = 'CFJ' then '吹风机'
when i.product_classification = 'LLJ' then '料理机'
when i.product_classification = 'MYY' then '美眼仪'
when i.product_classification = 'MFS' then '美发梳'
when i.product_classification = 'OTHER' then '其他'
end) 商品类型,
(case when s.shop_type = 'F' then '正品' else '样机' end) 产品类型,i.product_name 商品名称,i.count 明细数量,i.sn SN码,
(case when s.refund_status = 'ALL_REFUND' then '全部退货' 
when s.refund_status = 'PART_REFUND' then '部分退货' end) 退货状态,
(case when s.audit_status = 'WAIT_AUDIT' then '待审核'
when s.audit_status = 'SUCCESS' then '审核成功'
when s.audit_status = 'FAIL' then '审核失败' end) 审核状态,s.audit_remark 审核备注,
refund.count 退货数量,s.pay_amount 小计,gift.sg 标配赠品,gift.cg 条件赠品,'' 套购赠品
from sp_sales_order_item i left join sp_sales_order s on s.order_no = i.sales_order_no
join tmp_store st on s.store_id = st.store_id
left join tmp_refund refund on s.order_no = refund.sales_order_no
left join tmp_gift gift on s.order_no = gift.order_no
where s.id >= 63370 and s.id<= 100000 order by order_time
```
