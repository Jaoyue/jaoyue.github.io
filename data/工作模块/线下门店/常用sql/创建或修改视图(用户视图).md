select * from v_system_user 

CREATE OR REPLACE VIEW `platform-store`.`v_system_user` AS select concat('STAFF',`staff`.`id`) AS `user_id`,`staff`.`work_no` AS `work_no`,`staff`.`name` AS `name`,`staff`.`user_name` AS `user_name`,`staff`.`sales_region` AS `sales_region`,`staff`.`position` AS `position`,'STAFF' AS `type`,NULL AS `belong_region`,NULL AS `province_id`,NULL AS `city_id`,`staff`.`staff_status` AS `status` from `sp_biz_staff` `staff` where (1 = 1) group by `staff`.`id` union all select concat('CLERK',`clerk`.`id`) AS `user_id`,`clerk`.`clerk_no` AS `work_no`,`clerk`.`name` AS `name`,`clerk`.`account` AS `user_name`,group_concat(distinct `si`.`sales_region` separator ',') AS `sales_region`,`clerk`.`position` AS `position`,'CLERK' AS `type`,`vsr`.`store_region` AS `belong_region`,`si`.`province_id` AS `province_id`,`si`.`city_id` AS `city_id`,(case when (`clerk`.`delete_flag` = 0) then 'ENABLED' when (`clerk`.`delete_flag` = 1) then 'DISABLED' end) AS `status` from (((`sp_clerk_info` `clerk` left join `sp_clerk_store` `cs` on((`cs`.`clerk_id` = `clerk`.`id`))) left join `sp_store_info` `si` on((`si`.`id` = `cs`.`store_id`))) left join `v_store_region` `vsr` on((`vsr`.`store_id` = `cs`.`store_id`))) where (1 = 1) group by `clerk`.`id`;
