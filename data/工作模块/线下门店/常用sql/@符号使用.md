## 排名sql
使用@符号生成自增列给数据排名

```sql
SELECT TABALL.num,TABALL.year
FROM(
SELECT
		@row_number :=
	CASE
		
		WHEN @customer_no = TAB.year THEN
		@row_number + 1 ELSE 1 
		END AS num,
		@customer_no := TAB.year AS year,
		TAB.brand 
	FROM
		(
SELECT rw.brand,left(rw.month,2) year,sum(rw.retail_sales) AS retail_sales
FROM
sp_bi_off_retail_monthly rw 
WHERE rw.delete_flag = 0 
GROUP BY left(rw.month,2),rw.brand
ORDER BY year DESC,sum(rw.retail_sales) DESC
)TAB, (SELECT @row_number := 0) r,
 (SELECT @customer_no := '') r2 
)TABALL
WHERE TABALL.brand = '添可'
```
