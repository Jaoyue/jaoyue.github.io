```plsql

select * from sp_store_info


SELECT id, name, lbs,
    SUBSTRING_INDEX(lbs, ',', 1),SUBSTRING_INDEX(SUBSTRING_INDEX(lbs, ',', -1), ',', 1)
FROM sp_store_info;


SELECT
    *,
    6378.138 * 2 * ASIN(
      SQRT(
        POW(
          SIN(
            (
              '31.194689' * PI() / 180 - SUBSTRING_INDEX(SUBSTRING_INDEX(lbs, ',', -1), ',', 1) * PI() / 180
            ) / 2
          ), 2
        ) + COS('31.194689' * PI() / 180) * COS(SUBSTRING_INDEX(SUBSTRING_INDEX(lbs, ',', -1), ',', 1) * PI() / 180) * POW(
          SIN(
            (
              '120.604716' * PI() / 180 - SUBSTRING_INDEX(lbs, ',', 1) * PI() / 180
            ) / 2
          ), 2
        )
      )
    ) AS distance
FROM
    sp_store_info where lbs is not null
ORDER BY
    distance ASC
```
