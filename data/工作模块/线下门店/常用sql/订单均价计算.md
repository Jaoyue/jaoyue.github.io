```plsql
create table sp_sales_order_item_temp1 as
select i.id, 
(((select sum(order_amount) from sp_sales_order where order_no = i.sales_order_no)/(select sum(price) from sp_sales_order_item where sales_order_no = i.sales_order_no))*price) new_a
from sp_sales_order_item i
where i.average_price is null;

update sp_sales_order_item i1,sp_sales_order_item_temp1 i2
set i1.average_price = i2.new_a
where i1.id = i2.id ;

drop table sp_sales_order_item_temp1;
```
