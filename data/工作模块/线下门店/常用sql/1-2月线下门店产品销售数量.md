```plsql
select r.* from (
select 
s.name 店铺名称,a.area_name 省,a1.area_name 市,a2.area_name 区,t.material_no 产品编码,p.name 产品名称,t.count 台数
from(
select o.store_id,o.customer_province_id,o.customer_city_id,o.customer_district_id,i.material_no,sum(i.count) count from 
sp_sales_order_item i
join sp_sales_order o on i.sales_order_no = o.order_no
where o.create_time >= 1672502400000 and o.create_time < 1677600000000
GROUP BY o.store_id,o.customer_province_id,o.customer_city_id,o.customer_district_id,i.material_no
) t
join sp_store_info s on t.store_id = s.id
join sp_product p on t.material_no = p.material_no
join `platform-admin`.`sys_area` a on t.customer_province_id = a.area_id
join `platform-admin`.`sys_area` a1 on t.customer_city_id = a1.area_id
join `platform-admin`.`sys_area` a2 on t.customer_district_id = a2.area_id
 ) r order by r.店铺名称,r.省,r.市,r.区,r.产品编码


 
```
