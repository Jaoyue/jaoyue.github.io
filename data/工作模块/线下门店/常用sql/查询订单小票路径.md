```plsql
-- 根据年分组
select year,GROUP_CONCAT(file_path) from(

-- 查询小票全路径
select t.order_no order_no,right(i.path,30) name,i.path path,SUBSTR(i.path,21) file_path,SUBSTR(i.path,21,4) year  from `platform-fileserver`.`fs_file_info` i join (
select order_no,right(url,21) url from sp_order_voucher where order_no in (
'601633686770783047252'
)
) t on i.file_id = t.url

) r group by year order by year
```
