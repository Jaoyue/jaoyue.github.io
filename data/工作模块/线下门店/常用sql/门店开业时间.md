```plsql
-- 最晚开场
select i.store_no,i.name,FROM_UNIXTIME(o.op_time/1000,'%Y-%m-%d %H:%i:%s') 开业时间,FROM_UNIXTIME(i.retreat_time/1000,'%Y-%m-%d %H:%i:%s') 撤场时间  from sp_store_info i join (
select store_id,max(open_business_time) op_time from sp_store_decoration_life_cycle group by store_id
) o on i.id = o.store_id
-- 最早开场
select i.store_no,i.name,FROM_UNIXTIME(o.op_time/1000,'%Y-%m-%d %H:%i:%s') 开业时间,FROM_UNIXTIME(i.retreat_time/1000,'%Y-%m-%d %H:%i:%s') 撤场时间  from sp_store_info i join (
select store_id,min(open_business_time) op_time from sp_store_decoration_life_cycle group by store_id
) o on i.id = o.store_id

select * from(
select store_no,name,GROUP_CONCAT(最早开业时间) 最早开业时间,GROUP_CONCAT(最新开业时间) 最新开业时间,撤场时间 from(
select i.store_no,i.name,null 最早开业时间,FROM_UNIXTIME(o.op_time/1000,'%Y-%m-%d %H:%i:%s') 最新开业时间,FROM_UNIXTIME(i.retreat_time/1000,'%Y-%m-%d %H:%i:%s') 撤场时间  from sp_store_info i join (
select store_id,max(open_business_time) op_time from sp_store_decoration_life_cycle where delete_flag = 0 group by store_id
) o on i.id = o.store_id
union all 
select i.store_no,i.name,FROM_UNIXTIME(o.op_time/1000,'%Y-%m-%d %H:%i:%s') 最早开业时间,null 最新开业时间,FROM_UNIXTIME(i.retreat_time/1000,'%Y-%m-%d %H:%i:%s') 撤场时间  from sp_store_info i join (
select store_id,min(open_business_time) op_time from sp_store_decoration_life_cycle where delete_flag = 0 group by store_id
) o on i.id = o.store_id
) t group by store_no) t where 1=1 

HAVING 最早开业时间 <> 最新开业时间


select store_id,max(open_business_time) op_time from sp_store_decoration_life_cycle where store_id = 6493 group by store_id
select * from sp_store_decoration_life_cycle where store_id = 6493 group by store_id

select name,store_no,delete_flag from sp_store_info where negative_inventory_flag = 0
select * from sp_store_info where id = 6281
select * from sp_product where id = 6281
select * from sp_prototype_level where id = 6281

select l.id 编号,l.material_no 物料号,p.name,p.product_no,l.level_flag 等级,l.life_days 生命期,l.exceed_days 逾期,days_low 下限,days_up 上限 from sp_prototype_level l join sp_product p on l.material_no = p.material_no

select store_no 编码,name 名称,FROM_UNIXTIME(entering_time/1000,'%Y-%m-%d %H:%i:%s') 进场时间,acreage 面积,case 
when  exhibition_location = 'NEAR_DYSON' then '紧邻戴森' 
when  exhibition_location = 'ONE_LOCATION' then '小家电第1顺位' 
when  exhibition_location = 'TUO_LOCATION' then '中岛' 
when  exhibition_location = 'THREE_LOCATION' then '边厅' 
when  exhibition_location = 'FOUR_LOCATION' then '专厅' 
when  exhibition_location = 'FIVE_LOCATION' then '清洁电器第1顺位' 
when  exhibition_location = 'OTHER' then '其他' end 展厅位置,delete_flag 是否禁用
from sp_store_info order by delete_flag 














select store_id,name,GROUP_CONCAT(最早开业时间) 最早开业时间,GROUP_CONCAT(最新开业时间) 最新开业时间,撤场时间 from(
select i.id store_id,i.name,null 最早开业时间,FROM_UNIXTIME(o.op_time/1000,'%Y-%m-%d %H:%i:%s') 最新开业时间,FROM_UNIXTIME(i.retreat_time/1000,'%Y-%m-%d %H:%i:%s') 撤场时间  from sp_store_info i join (
select store_id,max(open_business_time) op_time from sp_store_decoration_life_cycle group by store_id
) o on i.id = o.store_id
union all 
select i.id store_id,i.name,FROM_UNIXTIME(o.op_time/1000,'%Y-%m-%d %H:%i:%s') 最早开业时间,null 最新开业时间,FROM_UNIXTIME(i.retreat_time/1000,'%Y-%m-%d %H:%i:%s') 撤场时间  from sp_store_info i join (
select store_id,min(open_business_time) op_time from sp_store_decoration_life_cycle group by store_id
) o on i.id = o.store_id
) t group by store_id 



select i.id store_id,i.name,null 最早开业时间,FROM_UNIXTIME(o.op_time/1000,'%Y-%m-%d %H:%i:%s') 最新开业时间,FROM_UNIXTIME(o.rest_time/1000,'%Y-%m-%d %H:%i:%s') 重置时间,FROM_UNIXTIME(i.retreat_time/1000,'%Y-%m-%d %H:%i:%s') 撤场时间 from sp_store_info i left join (
select store_id,max(open_business_time) op_time,rest_time from sp_store_decoration_life_cycle group by store_id
) o on i.id = o.store_id order by i.create_time desc
```
