```plsql
SELECT CONCAT_WS('',C.COLUMN_NAME,' ',C.COLUMN_COMMENT),T.table_name,T.TABLE_COMMENT ,C.COLUMN_NAME, C.COLUMN_COMMENT
FROM information_schema.TABLES T
        INNER JOIN
    information_schema.COLUMNS C ON T.TABLE_NAME = C.TABLE_NAME
WHERE C.TABLE_SCHEMA = 'platform-store' AND C.TABLE_NAME = 'sp_clerk_star_change';
```
