-- 查询考试分数
select 
e.id '序号',e.name '考试题目',
v.work_no '工号',u.staff_name '姓名',u.score '分数',ifnull(s.name,'无') '门店'
from sp_train_exam e
left join sp_train_user_exam u on e.id = u.exam_id
left join sp_store_info s on s.id = u.store_id
left join v_system_user v on u.user_id = v.user_id
where u.exam_id in (98,99)

select * from sp_train_exam  -- 210

select c.phone 账号,e.staff_name 姓名,e.score 分数,FROM_UNIXTIME(e.start_time/1000,'%Y-%m-%d %h:%i:%s') 开始时间,FROM_UNIXTIME(e.end_time/1000,'%Y-%m-%d %h:%i:%s') 交卷时间,(e.end_time-e.start_time)/1000 答题时长,
(case when e.`status` = 'commit' then '已提交'
 else '未提交' end) 考试状态
from sp_train_user_exam e join sp_clerk_info c on substr(e.user_id,6) = c.id where e.exam_id = 210 and e.user_id like 'CLERK%'
union all
select c.user_name 账号,e.staff_name 姓名,e.score 分数,FROM_UNIXTIME(e.start_time/1000,'%Y-%m-%d %h:%i:%s') 开始时间,FROM_UNIXTIME(e.end_time/1000,'%Y-%m-%d %h:%i:%s') 交卷时间,(e.end_time-e.start_time)/1000 答题时长,
(case when e.`status` = 'commit' then '已提交'
 else '未提交' end) 考试状态
from sp_train_user_exam e join sp_biz_staff c on substr(e.user_id,6) = c.id where e.exam_id = 210 and e.user_id like 'STAFF%'

) t
