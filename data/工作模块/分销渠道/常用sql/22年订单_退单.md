when d.sale_area = 'SUNINGDDJ' then '苏宁大单机'

when d.sale_area = 'LANZHOU' then '兰州分部'
when d.sale_area = 'QINGDAO' then '青岛分部'
when d.sale_area = 'HAERBIN' then '哈尔滨分部'

when d.sale_area = 'SHIJIAZHUANG' then '石家庄分部'
when d.sale_area = 'KA' then '大客户' end
) 销售大区,
FROM_UNIXTIME(o.create_time/1000,'%Y-%m-%d %h:%i:%s') 创建时间,

select
i.receive_refund_order_no 退货单号,
o.sales_channel 渠道模式,
(case 
when o.return_type = 'REJECTION' then '拒收'
when o.return_type = 'WITHDRAWAL' then '撤点退货'
when o.return_type = 'DEFECTIVE' then '不良品退货'
when o.return_type = 'UNSALEABLE' then '网点退货（呆滞）'
when o.return_type = 'RECALL' then '质量召回'
when o.return_type = 'ASM_RECEIVE' then '售后退货'
when o.return_type = 'ONLY_RECEIVE' then '仅退货' end
) 退货类型,
'退货退款' 售后类型,
FROM_UNIXTIME(o.create_time/1000,'%Y-%m-%d %h:%i:%s') 创建日期,
d.distributor_name 分销商名称,
(case 
when o.status = 'WAIT_AUDIT' then '待客审'
when o.status = 'WAIT_LEADER_AUDIT' then '待主管审'
when o.status = 'WAIT_FINANCE_AUDIT' then '待财务审'
when o.status = 'PROCESSING' then '处理中'
when o.status = 'WAIT_RECEIVE' then '待收货'
when o.status = 'WAIT_CHECK' then '待稽查'
when o.status = 'WAIT_LEADER_AUDIT_CHECK' then '稽查后待主管审'
when o.status = 'WAIT_FINANCE_AUDIT_CHECK' then '稽查后待财务审'
when o.status = 'CANCEL' then '作废'
when o.status = 'EXCEPTION' then '异常作废'
when o.status = 'FINISHED' then '完成' end
) 退货单状态,
o.original_order_no 外部单号,
o.transport_company_name 退回物流,
o.transport_number 退回面单号,
i.material_no 物料号,
i.product_name 商品名称,
i.product_price 销售单价,
i.return_price 退货单价,
i.return_count 退货数量,
i.return_price*i.return_count 退款金额合计,
o.pre_pay_amount 退预存款,
o.rebate_amount 退折让
from dist_receive_refund_order_item i
join dist_receive_refund_order o on i.receive_refund_order_no = o.receive_refund_order_no
join dist_distributor d on o.distributor_code = d.distributor_code
where o.create_time >= 1640966400000 and o.create_time< 1672502400000
