select c.order_no,c.distributor_code,c.distributor_name,i.material_no,i.product_name,r.sn,if(r.`status`=0,'正常','异常') 状态,r.error_dis_code,r.error_dis_name from dist_inventory_check c

join dist_sn_record r on c.order_no = r.order_no
