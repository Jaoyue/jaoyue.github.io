-- 活动保障主表  
CREATE TABLE ops_activity_assurance (  
    id BIGINT AUTO_INCREMENT PRIMARY KEY,  
    activity_no VARCHAR(100) COMMENT '活动编号（ATHB开头）',  
    activity_name VARCHAR(200) COMMENT '活动名称',  
    activity_desc TEXT COMMENT '活动描述',  
    ai_activity_level VARCHAR(20) COMMENT 'AI活动等级（MAJOR_SUPPORT/REGULAR_SUPPORT）',  
    activity_level VARCHAR(20) COMMENT '人工活动等级',  
    assurance_dept VARCHAR(100) COMMENT '保障部门',  
    assurance_dept_ai VARCHAR(100) COMMENT 'AI保障部门',  
    assurance_leader VARCHAR(100) COMMENT '保障负责人',  
    assurance_leader_id VARCHAR(100) COMMENT '保障负责人域账号',  
    assurance_leader_ai VARCHAR(100) COMMENT 'AI保障负责人',  
    assurance_leader_id_ai VARCHAR(100) COMMENT 'AI保障负责人域账号',  
    activity_status VARCHAR(20) DEFAULT 'PENDING' COMMENT '活动状态',  
    submitter_name VARCHAR(100) COMMENT '提报人名称',  
    submitter_account VARCHAR(100) COMMENT '提报人域账号',  
    start_time DATETIME COMMENT '活动开始时间',  
    end_time DATETIME COMMENT '活动结束时间',  
    review_time DATETIME COMMENT '复盘时间',  
    ai_plan_text LONGTEXT COMMENT 'AI活动方案文本',  
    manual_plan_text LONGTEXT COMMENT '人工纠偏方案文本',  
    ai_plan_status VARCHAR(20) DEFAULT 'NOT_GENERATED' COMMENT 'AI方案状态',  
    ai_analysis_status VARCHAR(20) DEFAULT 'PENDING' COMMENT 'AI分析状态',  
    plan_guard_flag VARCHAR(20) DEFAULT 'NOT_GUARDED' COMMENT '方案守门标识',  
    plan_guard_status VARCHAR(20) DEFAULT 'NOT_GUARDED' COMMENT '方案守门状态',  
    ai_action_status VARCHAR(20) DEFAULT 'NOT_GENERATED' COMMENT 'AI行动计划状态',  
    task_platform_task_id VARCHAR(100) COMMENT '任务中心守门任务ID',  
    version INT DEFAULT 0 COMMENT '版本号（乐观锁）',  
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,  
    create_by VARCHAR(50) DEFAULT '',  
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,  
    update_by VARCHAR(50) DEFAULT '',  
    is_deleted TINYINT DEFAULT 0,  
    INDEX idx_activity_no (activity_no),  
    INDEX idx_activity_status (activity_status),  
    INDEX idx_submitter (submitter_account),  
    INDEX idx_assurance_leader (assurance_leader_id)  
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='活动保障主表';  
  
-- 活动时间轴表  
CREATE TABLE ops_activity_timeline (  
    id BIGINT AUTO_INCREMENT PRIMARY KEY,  
    activity_id BIGINT NOT NULL COMMENT '关联活动主表ID',  
    start_time DATETIME COMMENT '开始时间',  
    end_time DATETIME COMMENT '结束时间',  
    core_task_desc VARCHAR(500) COMMENT '核心任务说明',  
    timeline_type VARCHAR(20) COMMENT '类型（STARTUP/STRESS_TEST/WARMUP/BURST）',  
    sort_order INT DEFAULT 0 COMMENT '排序序号',  
    version INT DEFAULT 0 COMMENT '版本号（乐观锁）',  
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,  
    create_by VARCHAR(50) DEFAULT '',  
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,  
    update_by VARCHAR(50) DEFAULT '',  
    is_deleted TINYINT DEFAULT 0,  
    INDEX idx_activity_id (activity_id)  
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='活动时间轴表';
  
-- 活动行动计划表  
CREATE TABLE ops_activity_action_plan (  
    id BIGINT AUTO_INCREMENT PRIMARY KEY,  
    activity_id BIGINT NOT NULL COMMENT '关联活动主表ID',  
    title VARCHAR(200) COMMENT '标题',  
    category VARCHAR(50) COMMENT '计划分类',  
    objective TEXT COMMENT '目标',  
    measure TEXT COMMENT '措施',  
    responsible_person VARCHAR(100) COMMENT '责任人',  
    responsible_person_id VARCHAR(100) COMMENT '责任人ID（域账号）',  
    expected_completion_time DATETIME COMMENT '预计完成时间',  
    annotation TEXT COMMENT '批注内容',  
    guard_status VARCHAR(20) DEFAULT 'NOT_GUARDED' COMMENT '守门状态',  
    guard_type VARCHAR(20) COMMENT '守门类型（SILICON_GUARDED/CARBON_GUARDED）',  
    is_edited INT DEFAULT 0 COMMENT '是否经过人工编辑',  
    is_pushed INT DEFAULT 0 COMMENT '是否已推送任务中心',  
    is_annotated INT DEFAULT 0 COMMENT '是否已批注',  
    task_platform_task_id VARCHAR(100) COMMENT '任务中心任务ID',  
    task_status VARCHAR(20) DEFAULT 'NOT_STARTED' COMMENT '任务中心状态',  
    version INT DEFAULT 0 COMMENT '版本号（乐观锁）',  
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,  
    create_by VARCHAR(50) DEFAULT '',  
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,  
    update_by VARCHAR(50) DEFAULT '',  
    is_deleted TINYINT DEFAULT 0,  
    INDEX idx_activity_id (activity_id),  
    INDEX idx_guard_status (guard_status)  
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='活动行动计划表';  
  
-- 行动计划操作日志表  
CREATE TABLE ops_activity_action_plan_log (  
    id BIGINT AUTO_INCREMENT PRIMARY KEY,  
    action_plan_id BIGINT NOT NULL COMMENT '关联行动计划ID',  
    activity_id BIGINT NOT NULL COMMENT '关联活动主表ID',  
    operation_type VARCHAR(20) COMMENT '操作类型',  
    operation_desc VARCHAR(500) COMMENT '操作描述',  
    before_value TEXT COMMENT '变更前值（JSON）',  
    after_value TEXT COMMENT '变更后值（JSON）',  
    operator_account VARCHAR(100) COMMENT '操作人域账号',  
    operator_name VARCHAR(100) COMMENT '操作人姓名',  
    version INT DEFAULT 0 COMMENT '版本号（乐观锁）',  
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,  
    create_by VARCHAR(50) DEFAULT '',  
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,  
    update_by VARCHAR(50) DEFAULT '',  
    is_deleted TINYINT DEFAULT 0,  
    INDEX idx_action_plan_id (action_plan_id),  
    INDEX idx_activity_id (activity_id)  
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='活动行动计划操作日志表';  
  
-- 活动复盘表  
CREATE TABLE ops_activity_review (  
    id BIGINT AUTO_INCREMENT PRIMARY KEY,  
    activity_id BIGINT NOT NULL COMMENT '关联活动主表ID',  
    review_status VARCHAR(20) DEFAULT 'DRAFT' COMMENT '复盘状态',  
    related_fault_ids TEXT COMMENT '关联故障/问题ID列表（JSON数组）',  
    fault_detail_text LONGTEXT COMMENT '拼接后的故障详情文本（传给复盘工作流）',  
    ai_review_report LONGTEXT COMMENT 'AI复盘报告',  
    manual_review_report LONGTEXT COMMENT '人工编辑复盘报告',  
    current_version VARCHAR(20) COMMENT '当前版本号',  
    guard_flag VARCHAR(20) DEFAULT 'NOT_GUARDED' COMMENT '守门标识',  
    guard_type VARCHAR(20) COMMENT '守门类型',  
    is_edited INT DEFAULT 0 COMMENT '是否经过人工编辑',  
    review_time DATETIME COMMENT '复盘提交时间',  
    request_id VARCHAR(100) COMMENT 'AI请求ID',  
    version INT DEFAULT 0 COMMENT '版本号（乐观锁）',  
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,  
    create_by VARCHAR(50) DEFAULT '',  
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,  
    update_by VARCHAR(50) DEFAULT '',  
    is_deleted TINYINT DEFAULT 0,  
    INDEX idx_activity_id (activity_id)  
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='活动复盘表';  
  
-- 活动复盘版本表  
CREATE TABLE ops_activity_review_version (  
    id BIGINT AUTO_INCREMENT PRIMARY KEY,  
    review_id BIGINT NOT NULL COMMENT '关联复盘记录ID',  
    activity_id BIGINT NOT NULL COMMENT '关联活动主表ID',  
    version_no VARCHAR(20) COMMENT '版本号（V1, V2, ...）',  
    review_report LONGTEXT COMMENT '该版本复盘报告内容',  
    is_current INT DEFAULT 0 COMMENT '是否当前版本',  
    source VARCHAR(20) COMMENT '来源（AI_GENERATED/MANUAL_EDITED）',  
    operator_account VARCHAR(100) COMMENT '操作人域账号',  
    operator_name VARCHAR(100) COMMENT '操作人姓名',  
    version INT DEFAULT 0 COMMENT '版本号（乐观锁）',  
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,  
    create_by VARCHAR(50) DEFAULT '',  
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,  
    update_by VARCHAR(50) DEFAULT '',  
    is_deleted TINYINT DEFAULT 0,  
    INDEX idx_review_id (review_id),  
    INDEX idx_activity_id (activity_id)  
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='活动复盘版本表';
-- 活动AI方案版本表  
CREATE TABLE IF NOT EXISTS `ops_activity_plan_version` (  
    `id`                     BIGINT       NOT NULL AUTO_INCREMENT COMMENT '主键',  
    `activity_id`            BIGINT       NOT NULL COMMENT '关联活动主表ID',  
    `version_no`             VARCHAR(20)  NOT NULL COMMENT '版本号（V1, V2, ...）',  
    `ai_plan_text`           LONGTEXT     NULL     COMMENT 'AI活动方案文本',  
    `manual_plan_text`       LONGTEXT     NULL     COMMENT '人工纠偏方案文本',  
    `ai_activity_level`      VARCHAR(20)  NULL     COMMENT 'AI活动等级',  
    `activity_level`         VARCHAR(20)  NULL     COMMENT '人工活动等级',  
    `assurance_dept`         VARCHAR(100) NULL     COMMENT '保障部门',  
    `assurance_dept_ai`      VARCHAR(100) NULL     COMMENT 'AI保障部门',  
    `assurance_leader`       VARCHAR(100) NULL     COMMENT '保障负责人',  
    `assurance_leader_id`    VARCHAR(100) NULL     COMMENT '保障负责人域账号',  
    `assurance_leader_ai`    VARCHAR(100) NULL     COMMENT 'AI保障负责人',  
    `assurance_leader_id_ai` VARCHAR(100) NULL     COMMENT 'AI保障负责人域账号',  
    `is_current`             INT          NOT NULL DEFAULT 0 COMMENT '是否当前版本：0否 1是',  
    `source`                 VARCHAR(20)  NULL     COMMENT '来源：AI_GENERATED / MANUAL_EDITED',  
    `operator_account`       VARCHAR(100) NULL     COMMENT '操作人域账号',  
    `operator_name`          VARCHAR(100) NULL     COMMENT '操作人姓名',  
     `version` INT DEFAULT 0 COMMENT '版本号（乐观锁）',  
    `create_time`            DATETIME     NULL     COMMENT '创建时间',  
    `create_by`              VARCHAR(100) NULL     COMMENT '创建人',  
    `update_time`            DATETIME     NULL     COMMENT '更新时间',  
    `update_by`              VARCHAR(100) NULL     COMMENT '更新人',  
    `is_deleted`             INT          NOT NULL DEFAULT 0 COMMENT '是否删除：0否 1是',  
    PRIMARY KEY (`id`),  
    INDEX `idx_activity_id` (`activity_id`)  
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='活动AI方案版本表';

create table ops_activity_review_fault  
(  
    id             bigint auto_increment  
        primary key,  
    activity_id    bigint                                not null comment '关联活动保障主表ID',  
    fault_id       bigint                                not null comment '关联故障ID',  
    fault_no       varchar(100)                          null comment '故障编码',  
    fault_desc     text                                  null comment '故障描述',  
    fault_level    varchar(50)                           null comment '故障等级',  
    product_name   varchar(200)                          null comment '产品名称',  
    product_module varchar(200)                          null comment '产品模块',  
    department     varchar(100)                          null comment '所属部门',  
    version        int         default 0                 null comment '版本号',  
    create_time    timestamp   default CURRENT_TIMESTAMP null,  
    create_by      varchar(50) default ''                null,  
    update_time    timestamp   default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP,  
    update_by      varchar(50) default ''                null,  
    is_deleted     tinyint     default 0                 null  
)  
    comment '活动复盘关联故障表' charset = utf8mb4;  
  
create index idx_activity_id  
    on ops_activity_review_fault (activity_id);  
  
create index idx_fault_id  
    on ops_activity_review_fault (fault_id);
