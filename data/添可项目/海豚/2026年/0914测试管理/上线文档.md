## 建表SQL

```

ALTER TABLE rev_test_case  
ADD COLUMN project_code VARCHAR(64) NULL COMMENT '项目编码' AFTER project_id,  
ADD COLUMN importance VARCHAR(16) NULL COMMENT '重要程度：HIGH-高 MEDIUM-中 LOW-低' AFTER priority,  
ADD COLUMN system_feature_code VARCHAR(64) NULL COMMENT '系统功能code（关联 pm_blueprint_system_feature.feature_code）' AFTER system_feature_id,  
ADD COLUMN module_code VARCHAR(32) NULL COMMENT '模块编号（关联 pm_dev_solution_module.module_code）' AFTER module_name;

alter table rev_test_report add column module varchar(500) null comment '模块名称' after `name`;

create table pm_test_plan  
(  
    id                 bigint auto_increment comment '主键ID'  
        primary key,  
    code               varchar(32)                              not null comment '测试计划编码 ATTP+yyMMdd+4位（决策3）',  
    project_id         varchar(64)                              null comment '项目ID',  
    project_name       varchar(128)                             null comment '项目名称',  
    project_code       varchar(64)                              null comment '项目编码',  
    product_id         bigint                                   null comment '产品ID',  
    name               varchar(256)                             null comment '计划名称',  
    content            longtext                                 null comment '计划内容(Markdown)',  
    ai_content         longtext                                 null comment 'AI原文',  
    final_content      longtext                                 null comment '人工终稿',  
    solution_id        bigint                                   null comment '关联测试方案ID（需求变更：方案在前计划在后）',  
    solution_code      varchar(32)                              null comment '关联测试方案编码',  
    personnel_file_url varchar(512)                             null comment '人员清单文件链接（前端上传后URL，工作流自行解析，计划生成依据）',  
    status             varchar(32) default 'WAITING_GATEKEEPER' null comment 'WAITING_GATEKEEPER-待守门 CONFIRMED-已守门',  
    gatekeeper_id      varchar(64)                              null comment '守门人ID',  
    gatekeeper_name    varchar(64)                              null comment '守门人姓名',  
    confirmed_at       datetime                                 null comment '守门确认时间',  
    creator_id         varchar(64)                              null comment '创建人ID',  
    creator_name       varchar(64)                              null comment '创建人姓名',  
    version            int         default 0                    not null comment '版本号（乐观锁，BaseEntity @Version）',  
    create_time        datetime    default CURRENT_TIMESTAMP    not null comment '创建时间',  
    update_time        datetime    default CURRENT_TIMESTAMP    not null on update CURRENT_TIMESTAMP comment '更新时间',  
    create_by          varchar(64)                              null comment '创建人账号（BaseEntity 自动填充）',  
    update_by          varchar(64)                              null comment '更新人账号（BaseEntity 自动填充）',  
    is_deleted         tinyint(1)  default 0                    not null comment '是否删除 0-否 1-是（BaseEntity @TableLogic）',  
    constraint uk_code  
        unique (code)  
)  
    comment '项目测试计划表' charset = utf8mb4;  
  
create index idx_project_code  
    on pm_test_plan (project_code);  
  
create index idx_status  
    on pm_test_plan (status);

create table pm_test_solution  
(  
    id              bigint auto_increment comment '主键ID'  
        primary key,  
    code            varchar(32)                              not null comment '测试方案编码 ATTS+yyMMdd+4位（决策3）',  
    project_id      varchar(64)                              null comment '项目ID',  
    project_name    varchar(128)                             null comment '项目名称',  
    project_code    varchar(64)                              null comment '项目编码',  
    product_id      bigint                                   null comment '产品ID',  
    name            varchar(256)                             null comment '方案名称',  
    content         longtext                                 null comment '方案内容(Markdown)',  
    ai_content      longtext                                 null comment 'AI原文',  
    final_content   longtext                                 null comment '人工终稿',  
    plan_id         bigint                                   null comment '关联测试计划ID',  
    plan_code       varchar(32)                              null comment '关联测试计划编码',  
    status          varchar(32) default 'WAITING_GATEKEEPER' null comment 'WAITING_GATEKEEPER-待守门 CONFIRMED-已守门',  
    gatekeeper_id   varchar(64)                              null comment '守门人ID',  
    gatekeeper_name varchar(64)                              null comment '守门人姓名',  
    confirmed_at    datetime                                 null comment '守门确认时间',  
    creator_id      varchar(64)                              null comment '创建人ID',  
    creator_name    varchar(64)                              null comment '创建人姓名',  
    version         int         default 0                    not null comment '版本号（乐观锁，BaseEntity @Version）',  
    create_time     datetime    default CURRENT_TIMESTAMP    not null comment '创建时间',  
    update_time     datetime    default CURRENT_TIMESTAMP    not null on update CURRENT_TIMESTAMP comment '更新时间',  
    create_by       varchar(64)                              null comment '创建人账号（BaseEntity 自动填充）',  
    update_by       varchar(64)                              null comment '更新人账号（BaseEntity 自动填充）',  
    is_deleted      tinyint(1)  default 0                    not null comment '是否删除 0-否 1-是（BaseEntity @TableLogic）',  
    constraint uk_code  
        unique (code)  
)  
    comment '项目测试方案表' charset = utf8mb4;  
  
create index idx_project_code  
    on pm_test_solution (project_code);  
  
create index idx_status  
    on pm_test_solution (status);
```
