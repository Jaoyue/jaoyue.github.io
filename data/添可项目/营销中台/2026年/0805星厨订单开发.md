# 星厨订单开发

## 大货赠品关系管理

### 建表
```sql
-- Vx.y__IMS_PRODUCT_GIFT_RELATION_ADD.sql
CREATE TABLE `ims_product_gift_relation` (
  `relation_id`       bigint(20)   NOT NULL    COMMENT '主键ID',
  `goods_sap_id`      varchar(64)  NOT NULL DEFAULT '' COMMENT '大货SAP ID',
  `goods_name`        varchar(255) NOT NULL DEFAULT '' COMMENT '大货名称',
  `gift_sap_id`       varchar(64)  NOT NULL DEFAULT '' COMMENT '赠品SAP ID',
  `gift_name`         varchar(255) NOT NULL DEFAULT '' COMMENT '赠品名称',
  `version_num`       int(11)      NOT NULL DEFAULT '0' COMMENT '版本号',
  `delete_flag`       tinyint(1)   NOT NULL DEFAULT '0' COMMENT '删除标识 1-删除 0-未删除',
  `creation_date`     datetime     DEFAULT NULL COMMENT '创建时间',
  `created_by`        int(11)      NOT NULL DEFAULT '-1' COMMENT '创建人',
  `last_updated_by`   int(11)      NOT NULL DEFAULT '-1' COMMENT '最后更新人',
  `last_update_date`  datetime     DEFAULT NULL COMMENT '最后更新时间',
  `last_update_login` int(11)      NOT NULL DEFAULT '-1' COMMENT '最后登录人',
  PRIMARY KEY (`relation_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='大货赠品关系表';
```
