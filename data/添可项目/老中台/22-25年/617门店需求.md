### 门店客户记录
```sql
CREATE TABLE `platform-store`.`sp_store_company_record`  (
  `id` bigint(20) NOT NULL AUTO_INCREMENT COMMENT '主键,自增',
  `delete_flag` int(11) NOT NULL COMMENT '是否被删除，0：否  1：是',
  `creater` varchar(50) CHARACTER SET utf8 COLLATE utf8_general_ci NOT NULL COMMENT '创建者',
  `create_time` bigint(20) NOT NULL COMMENT '创建时间',
  `modifier` varchar(50) CHARACTER SET utf8 COLLATE utf8_general_ci NOT NULL COMMENT '更新者',
  `modify_time` bigint(20) NOT NULL COMMENT '更新时间',
  `version` bigint(20) NOT NULL,
  `store_id` bigint(20) NOT NULL COMMENT '门店ID',
  `start_time` bigint(20) NULL COMMENT '起始日期',
  `end_time` bigint(20) NULL COMMENT '终止日期',
  `company_no` varchar(255) NULL COMMENT '客户编码',
  `company_name` varchar(255) NULL COMMENT '客户名称',
  PRIMARY KEY (`id`)
);
```
