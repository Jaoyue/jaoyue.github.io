## SQL
```
CREATE TABLE `sp_user_stage_assessment_report` (  
  `id` BIGINT(20) NOT NULL AUTO_INCREMENT COMMENT '主键id',  
  `account` VARCHAR(64) DEFAULT NULL COMMENT '工号',  
  `record_staff_id` BIGINT(20) DEFAULT NULL COMMENT '记录人员id（根据工号关联查询到的业务员id）',  
  `name` VARCHAR(255) DEFAULT NULL COMMENT '报告名称',  
  `url` VARCHAR(1024) DEFAULT NULL COMMENT '报告链接url',  
  `report_date` VARCHAR(32) DEFAULT NULL COMMENT '报告日期（yyyy-MM-dd）',  
  `creater` VARCHAR(64) DEFAULT NULL COMMENT '创建人',  
  `create_time` BIGINT(20) DEFAULT NULL COMMENT '创建时间（毫秒时间戳）',  
  `modifier` VARCHAR(64) DEFAULT NULL COMMENT '修改人',  
  `modify_time` BIGINT(20) DEFAULT NULL COMMENT '修改时间（毫秒时间戳）',  
  `version` BIGINT(20) DEFAULT NULL COMMENT '版本号',  
  PRIMARY KEY (`id`),  
  KEY `idx_account` (`account`),  
  KEY `idx_record_staff_id` (`record_staff_id`)  
) COMMENT='用户阶段评估报告';
```
