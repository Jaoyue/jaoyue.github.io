# 1.市场合规导入

### sql

```
CREATE TABLE sp_market_violation
(
    id                  BIGINT AUTO_INCREMENT
        PRIMARY KEY COMMENT '主键ID',
    violation_type                VARCHAR(50) NOT NULL COMMENT '类型：市场违规、虚假订单等',
    company_no         VARCHAR(50) NOT NULL COMMENT '经销商编码',
    company_name         VARCHAR(200) NOT NULL COMMENT '经销商名称',
    sn                  VARCHAR(100) NOT NULL COMMENT '产品SN序列号',
    source_order_no     VARCHAR(100) NULL COMMENT '来源单号',
    file_no             VARCHAR(50) NULL COMMENT '文件号',
    occur_time          BIGINT NOT NULL COMMENT '发生时间',
    delete_flag         INT NOT NULL DEFAULT 0 COMMENT '是否删除：0-否 1-是',
    creater             VARCHAR(50) NOT NULL COMMENT '创建人',
    create_time         BIGINT NOT NULL COMMENT '创建时间',
    modifier            VARCHAR(50) NOT NULL COMMENT '修改人',
    modify_time         BIGINT NOT NULL COMMENT '修改时间',
    version             BIGINT NOT NULL DEFAULT 0 COMMENT '版本号',
    INDEX idx_dealer_code (company_no),
    INDEX idx_sn (sn),
    INDEX idx_occur_time (occur_time)
) COMMENT '市场合规表';
```
