import json

def convert_string_to_json_array(input_string, arg2):
    # 以逗号分割字符串
    items = input_string.split(',')
    
    result = []
    second_level_faults = []
    third_level_faults = []

    for item in items:
        # 以中划线分割每个项目
        parts = item.split('-')
        
        obj = {}
        if len(parts) == 1:
            # 只有一个部分，视为三级故障
            third_level = parts[0].strip()
            second_level_faults.append(third_level)
            third_level_faults.append(third_level)
            
            obj = {
                "三级故障": third_level
            }
        elif len(parts) == 2:
            # 有两个部分，视为二级和三级故障
            second_level = parts[0].strip()
            third_level = parts[1].strip()
            
            second_level_faults.append(second_level)
            third_level_faults.append(third_level)

            obj = {
                "二级故障": second_level,
                "三级故障": third_level
            }
        elif len(parts) >= 3:
            # 有三个或更多部分，视为一级、二级和三级故障
            second_level = parts[1].strip()
            third_level = parts[2].strip()
            
            second_level_faults.append(second_level)
            third_level_faults.append(third_level)
            
            obj = {
                "一级故障": parts[0].strip(),
                "二级故障": second_level,
                "三级故障": third_level
            }
        
        result.append(obj)
    
    # 将去重后的二级故障用逗号拼接
    unique_second_level_faults = ",".join(second_level_faults)
    # 如果二级故障为空，则使用三级故障
    if not unique_second_level_faults:
        unique_second_level_faults = ",".join(third_level_faults)
    # 将去重后的三级故障用逗号拼接
    unique_third_level_faults = ",".join(third_level_faults)
    
    if arg2 == 'tineco':
        unique_second_level_faults = input_string
        unique_third_level_faults = input_string
    if arg2 == 'ecovacs':
        unique_third_level_faults = input_string

    return result, unique_second_level_faults, unique_third_level_faults

def main(arg1: str, arg2: str) -> dict:
    # 尝试解析 JSON，如果失败则直接使用 arg1
    try:
        arg1_json = json.loads(arg1)
        content = arg1_json.get("content", arg1)  # 如果 content 不存在，则使用 arg1
    except json.JSONDecodeError:
        content = arg1  # 如果 arg1 不是 JSON，则直接使用
    
    # 调用原有逻辑
    result, second_level_faults, third_level_faults = convert_string_to_json_array(content, arg2)
    
    return {
        "result": {
            "second_level_faults": second_level_faults,
            "third_level_faults": third_level_faults
        },
    }
