#### 数据源配置类
:::warning
1.主数据源的配置非常关键

2.副数据源在使用事务时需指定<font style="color:rgb(245, 171, 53);background-color:rgb(28, 39, 58);">@Transactional("dmptinecoTransactionManager")</font>

:::

```java
package com.tineco.pyne.application.config;

import com.alibaba.druid.spring.boot3.autoconfigure.DruidDataSourceBuilder;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.boot.context.properties.ConfigurationProperties;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.datasource.DataSourceTransactionManager;
import org.springframework.transaction.PlatformTransactionManager;

import javax.sql.DataSource;

@Configuration
public class DataSourceConfig {

    // 主数据源（MyBatis-Plus 使用）
    @Bean
    @Primary // 关键注解
    @ConfigurationProperties(prefix = "spring.datasource")
    public DataSource primaryDataSource() {
        return DruidDataSourceBuilder.create().build();
    }
    // 主数据源事务管理器（关键！）
    @Primary
    @Bean
    public PlatformTransactionManager primaryTransactionManager(
            @Qualifier("primaryDataSource") DataSource dataSource) {
        return new DataSourceTransactionManager(dataSource);
    }

    // ========== dmptineco 数据源 ==========
    @Bean(name = "dmptinecoDataSource")
    @ConfigurationProperties(prefix = "spring.datasource.dmptineco")
    public DataSource dmptinecoDataSource() {
        return DruidDataSourceBuilder.create().build();
    }

    @Bean(name = "dmptinecoJdbcTemplate")
    public JdbcTemplate dmptinecoJdbcTemplate(
            @Qualifier("dmptinecoDataSource") DataSource dataSource) {
        return new JdbcTemplate(dataSource);
    }

    @Bean(name = "dmptinecoTransactionManager")
    public PlatformTransactionManager dmptinecoTransactionManager(
            @Qualifier("dmptinecoDataSource") DataSource dataSource) {
        return new DataSourceTransactionManager(dataSource);
    }

    // ========== dmpecovacs 数据源 ==========
    @Bean(name = "dmpecovacsDataSource")
    @ConfigurationProperties(prefix = "spring.datasource.dmpecovacs")
    public DataSource dmpecovacsDataSource() {
        return DruidDataSourceBuilder.create().build();
    }

    @Bean(name = "dmpecovacsJdbcTemplate")
    public JdbcTemplate dmpecovacsJdbcTemplate(
            @Qualifier("dmpecovacsDataSource") DataSource dataSource) {
        return new JdbcTemplate(dataSource);
    }

    @Bean(name = "dmpecovacsTransactionManager")
    public PlatformTransactionManager dmpecovacsTransactionManager(
            @Qualifier("dmpecovacsDataSource") DataSource dataSource) {
        return new DataSourceTransactionManager(dataSource);
    }
}


```

### 副数据源应用类(JdbcTemplate用法)
```java
package com.tineco.pyne.application.service.impl.aide;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Map;

/**
 * @date: 2025/8/12 13:46
 */
@Service
public class SecondaryDsService {
    @Autowired
    @Qualifier("dmptinecoJdbcTemplate")
    private JdbcTemplate dmptinecoJdbcTemplate;
    @Autowired
    @Qualifier("dmpecovacsJdbcTemplate")
    private JdbcTemplate dmpecovacsJdbcTemplate;

    public List<Map<String, Object>> queryDataFromDmptineco(String sql) {
        return dmptinecoJdbcTemplate.queryForList(sql);
    }

    public List<Map<String, Object>> queryDataFromDmpecovacs(String sql) {
        return dmpecovacsJdbcTemplate.queryForList(sql);
    }
}
```
