```java
package com.ecovacs.store.util.JGPush;

import java.util.Map;

/**
 * @date: 2024/11/27 16:23
 */
public class PushBean {
    // 必填, 通知内容, 内容可以为空字符串，则表示不展示到通知栏。
    private String alert;
    // 可选, 附加信息, 供业务使用。
    private Map<String, String> extras;
    //android专用// 可选, 通知标题	如果指定了，则通知里原来展示 App名称的地方，将展示成这个字段。
    private String title;
    public PushBean(String alert, Map<String, String> extras, String title) {
        this.alert = alert;
        this.extras = extras;
        this.title = title;
    }

    @Override
    public String toString() {
        return "PushBean{" +
                "alert='" + alert + '\'' +
                ", extras=" + extras +
                ", title='" + title + '\'' +
                '}';
    }
    public String getAlert() {
        return alert;
    }

    public void setAlert(String alert) {
        this.alert = alert;
    }

    public Map<String, String> getExtras() {
        return extras;
    }

    public void setExtras(Map<String, String> extras) {
        this.extras = extras;
    }

    public String getTitle() {
        return title;
    }

    public void setTitle(String title) {
        this.title = title;
    }
}
```

```java
package com.ecovacs.store.util.JGPush;

import cn.jiguang.common.resp.APIConnectionException;
import cn.jiguang.common.resp.APIRequestException;
import cn.jpush.api.JPushClient;
import cn.jpush.api.push.PushResult;
import cn.jpush.api.push.model.Platform;
import cn.jpush.api.push.model.PushPayload;
import cn.jpush.api.push.model.audience.Audience;
import cn.jpush.api.push.model.notification.Notification;
import com.ecovacs.store.cache.ConfigCacheManager;
import com.ecovacs.store.constants.MyConstants;
import org.apache.commons.lang.StringUtils;
import org.apache.log4j.Logger;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;


public class JGPushUtil {
    private static final Logger logger = Logger.getLogger(JGPushUtil.class);
    private static String appKey = null;
    private static String masterSecret = null;
    private static final int max_size = 3;


    /**
     * 广播 (所有平台，所有设备, 不支持附加信息)
     * @return
     */
    public static boolean pushAll(PushBean pushBean){
        return sendPush(Platform.all(),null,Notification.alert(pushBean.getAlert()));
    }

    /**
     * 推送全部ios ios广播
     * @return
     */
    public static boolean pushIos(PushBean pushBean){
        if (StringUtils.isNotBlank(pushBean.getTitle())){
            pushBean.setAlert(pushBean.getTitle() + "\n" + pushBean.getAlert());
        }
        return sendPush(Platform.ios(),null,Notification.ios(pushBean.getAlert(), pushBean.getExtras()));
    }

    /**
     * 推送ios 指定id
     * @return
     */
    public static boolean pushIos(PushBean pushBean, String... registids){
        if (StringUtils.isNotBlank(pushBean.getTitle())){
            pushBean.setAlert(pushBean.getTitle() + "\n" + pushBean.getAlert());
        }
        return sendPush(Platform.ios(),registids,Notification.ios(pushBean.getAlert(), pushBean.getExtras()));
    }

    /**
     * 推送全部android
     * @return
     */
    public static boolean pushAndroid(PushBean pushBean){
        return sendPush(Platform.android(),null,Notification.android(pushBean.getAlert(), pushBean.getTitle(), pushBean.getExtras()));
    }

    /**
     * 推送android 指定id
     * @return
     */
    public static boolean pushAndroid(PushBean pushBean, String... registids){
        return sendPush(Platform.android(),registids,Notification.android(pushBean.getAlert(), pushBean.getTitle(), pushBean.getExtras()));
    }

    private static boolean sendPush(Platform platform, String[] registids, Notification notification) {
        boolean pushFlag = true;
        List<String[]> registidList = getStringArr(registids);
        for (String[] registIds : registidList) {
            PushPayload payload = PushPayload.newBuilder()
                    .setPlatform(platform)
                    .setAudience(registids == null ? Audience.all() : Audience.registrationId(registids))
                    .setNotification(notification)
                    .build();
            masterSecret = ConfigCacheManager.getConfigValue(MyConstants.SP_JG_PUSH_MASTERSECRET);
            appKey = ConfigCacheManager.getConfigValue(MyConstants.SP_JG_PUSH_APPKEY);
            JPushClient jpushClient = new JPushClient(masterSecret, appKey);
            try {
                // 发送推送
                PushResult result = jpushClient.sendPush(payload);
                if (result!=null && result.isResultOK()){
                    pushFlag = true;
                }else {
                    pushFlag = false;
                }
                logger.info("Push result : " + result);
            } catch (APIConnectionException e) {
                logger.info("Connection error. Should retry later." + e);
                pushFlag = false;
            } catch (APIRequestException e) {
                logger.info("Error response from JPush server. Should review and fix it." + e);
                logger.info("HTTP Status: " + e.getStatus());
                logger.info("Error Code: " + e.getErrorCode());
                logger.info("Error Message: " + e.getErrorMessage());
                pushFlag = false;
            }
        }
        return pushFlag;
    }

    private static List<String[]> getStringArr(String[] registids) {
        if (registids == null || registids.length == 0) {
            // 返回一个size为1的list
            return new ArrayList<>(1);
        }
        List<String[]> chunks = new ArrayList<>();
        int length = registids.length;
        for (int i = 0; i < length; i += max_size) {
            int end = Math.min(length, i + max_size);
            String[] chunk = Arrays.copyOfRange(registids, i, end);
            chunks.add(chunk);
        }
        return chunks;
    }

}
```
