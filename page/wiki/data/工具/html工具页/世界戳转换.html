private void OpenConsole()
{
    // 路径包裹双引号，处理特殊字符
    ProcessStartInfo psi = new ProcessStartInfo("cmd.exe", string.Format("/K cd /d \"{0}\"", rootFolderPath));
    Process.Start(psi);
}

private void ExecuteGitCommand(string command)
{
    // 路径包裹双引号，处理特殊字符
    ProcessStartInfo psi = new ProcessStartInfo("cmd.exe", string.Format("/C cd /d \"{0}\" && {1}", rootFolderPath, command));
    psi.UseShellExecute = false;
    psi.RedirectStandardOutput = true;
    psi.RedirectStandardError = true;
    psi.CreateNoWindow = true;

    using (Process process = Process.Start(psi))
    {
        string output = process.StandardOutput.ReadToEnd();
        string error = process.StandardError.ReadToEnd();
        process.WaitForExit();

        if (!string.IsNullOrEmpty(output))
        {
            MessageBox.Show(output, "Git Output", MessageBoxButtons.OK, MessageBoxIcon.Information);
        }
        if (!string.IsNullOrEmpty(error))
        {
            MessageBox.Show(error, "Git Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
        }
    }
}

private void GitPush()
{
    string commitMessage = PromptForCommitMessage();
    if (string.IsNullOrEmpty(commitMessage))
    {
        return;
    }

    ExecuteGitCommand("git add .");
    ExecuteGitCommand(string.Format("git commit -m \"{0}\"", commitMessage));
    ExecuteGitCommand("git push");
}

// 重写快捷键处理方法，新增Ctrl+C、Ctrl+X、Ctrl+V处理
protected override bool ProcessCmdKey(ref Message msg, Keys keyData)
{
    if (keyData == Keys.Escape)
    {
        this.WindowState = FormWindowState.Minimized;
        return true;
    }
    // 处理Ctrl+C复制
    else if (keyData == (Keys.Control | Keys.C))
    {
        CopyItem();
        return true;
    }
    // 处理Ctrl+X剪切
    else if (keyData == (Keys.Control | Keys.X))
    {
        CutItem();
        return true;
    }
    // 处理Ctrl+V粘贴
    else if (keyData == (Keys.Control | Keys.V))
    {
        PasteItem();
        return true;
    }
    return base.ProcessCmdKey(ref msg, keyData);
}

// 复制选中的文件/文件夹
private void CopyItem()
{
    TreeNode selectedNode = treeViewDirectory.SelectedNode;
    if (selectedNode == null || selectedNode.Tag == null)
    {
        ShowNotification("请选择要复制的文件或文件夹！", Color.Orange); // 橙色提示警告
        return;
    }

    string path = selectedNode.Tag.ToString();
    if (File.Exists(path) || Directory.Exists(path))
    {
        copiedItemPath = path;
        isCut = false; // 复制操作，剪切标记置为false
        // 也可以将路径写入系统剪贴板，增强兼容性
        Clipboard.SetText(path);
        ShowNotification(string.Format("已复制：{0}", path)); // 绿色提示成功
    }
    else
    {
        ShowNotification("选中的项不是有效的文件或文件夹！", Color.Orange);
    }
}

// 剪切选中的文件/文件夹
private void CutItem()
{
    TreeNode selectedNode = treeViewDirectory.SelectedNode;
    if (selectedNode == null || selectedNode.Tag == null)
    {
        ShowNotification("请选择要剪切的文件或文件夹！", Color.Orange);
        return;
    }

    string path = selectedNode.Tag.ToString();
    if (File.Exists(path) || Directory.Exists(path))
    {
        copiedItemPath = path;
        isCut = true; // 剪切操作，剪切标记置为true
        Clipboard.SetText(path);
        ShowNotification(string.Format("已剪切：{0}", path));
    }
    else
    {
        ShowNotification("选中的项不是有效的文件或文件夹！", Color.Orange);
    }
}

// 粘贴文件/文件夹到选中的文件夹
private void PasteItem()
{
    // 检查是否有复制/剪切的内容
    if (string.IsNullOrEmpty(copiedItemPath) && string.IsNullOrEmpty(Clipboard.GetText()))
    {
        ShowNotification("没有可粘贴的内容！", Color.Orange);
        return;
    }

    // 优先使用本地变量，没有则从系统剪贴板获取
    string sourcePath = !string.IsNullOrEmpty(copiedItemPath) ? copiedItemPath : Clipboard.GetText();
    if (!File.Exists(sourcePath) && !Directory.Exists(sourcePath))
    {
        ShowNotification("剪贴板中的内容不是有效的文件或文件夹路径！", Color.Orange);
        return;
    }

    // 检查选中的目标位置
    TreeNode selectedNode = treeViewDirectory.SelectedNode;
    if (selectedNode == null || selectedNode.Tag == null)
    {
        ShowNotification("请选择要粘贴到的目标文件夹！", Color.Orange);
        return;
    }

    string targetPath = selectedNode.Tag.ToString();
    // 如果选中的是文件，则取其所在目录作为目标路径
    if (File.Exists(targetPath))
    {
        targetPath = Path.GetDirectoryName(targetPath);
    }
    // 确保目标路径是文件夹
    if (!Directory.Exists(targetPath))
    {
        ShowNotification("目标位置不是有效的文件夹！", Color.Orange);
        return;
    }

    // 防止粘贴到自身目录（文件夹）
    if (Directory.Exists(sourcePath) && targetPath.StartsWith(sourcePath, StringComparison.OrdinalIgnoreCase))
    {
        ShowNotification("无法将文件夹粘贴到自身目录中！", Color.Orange);
        return;
    }

    // 构建目标文件/文件夹的路径
    string itemName = Path.GetFileName(sourcePath);
    string destPath = Path.Combine(targetPath, itemName);

    // 处理重名问题
    destPath = GetUniquePath(destPath);

    try
    {
        if (File.Exists(sourcePath))
        {
            // 复制文件
            File.Copy(sourcePath, destPath, false); // false：不覆盖已存在的文件
            // 如果是剪切操作，复制后删除原文件
            if (isCut)
            {
                File.Delete(sourcePath);
                ShowNotification(string.Format("文件已剪切到：{0}", destPath));
            }
            else
            {
                ShowNotification(string.Format("文件已粘贴到：{0}", destPath));
            }
        }
        else if (Directory.Exists(sourcePath))
        {
            // 复制文件夹（递归复制所有内容）
            CopyDirectory(sourcePath, destPath);
            // 如果是剪切操作，复制后删除原文件夹
            if (isCut)
            {
                Directory.Delete(sourcePath, true);
                ShowNotification(string.Format("文件夹已剪切到：{0}", destPath));
            }
            else
            {
                ShowNotification(string.Format("文件夹已粘贴到：{0}", destPath));
            }
        }

        // 记录目标路径的父级，刷新后定位
        string parentPath = Path.GetDirectoryName(destPath);
        RefreshTreeView(parentPath);
        SelectFileInTreeView(destPath);

        // 剪切操作完成后，重置剪切标记（可选，避免重复剪切）
        if (isCut)
        {
            isCut = false;
            copiedItemPath = null;
        }
    }
    catch (UnauthorizedAccessException)
    {
        ShowNotification("没有权限执行粘贴操作！", Color.Red); // 红色提示错误
    }
    catch (Exception ex)
    {
        ShowNotification(string.Format("粘贴失败：{0}", ex.Message), Color.Red);
    }
}

// 获取唯一的路径（处理重名，自动加数字后缀）
private string GetUniquePath(string path)
{
    if (!File.Exists(path) && !Directory.Exists(path))
    {
        return path;
    }

    string directory = Path.GetDirectoryName(path);
    string fileName = Path.GetFileNameWithoutExtension(path);
    string extension = Path.GetExtension(path);
    int index = 1;

    // 循环直到找到不重复的路径
    do
    {
        string newName = string.Format("{0}({1}){2}", fileName, index, extension);
        path = Path.Combine(directory, newName);
        index++;
    } while (File.Exists(path) || Directory.Exists(path));

    return path;
}

// 递归复制文件夹
private void CopyDirectory(string sourceDir, string destDir)
{
    // 创建目标文件夹
    Directory.CreateDirectory(destDir);

    // 复制文件夹中的文件
    foreach (string file in Directory.GetFiles(sourceDir))
    {
        string destFile = Path.Combine(destDir, Path.GetFileName(file));
        File.Copy(file, destFile, false);
    }

    // 递归复制子文件夹
    foreach (string subDir in Directory.GetDirectories(sourceDir))
    {
        string destSubDir = Path.Combine(destDir, Path.GetFileName(subDir));
        CopyDirectory(subDir, destSubDir);
    }
}

private void TreeViewDirectory_DrawNode(object sender, DrawTreeNodeEventArgs e)
{
    Font nodeFont = treeViewDirectory.Font;
    Color textColor = e.Node.ForeColor;
    Color backgroundColor = SystemColors.Window;

    if (highlightedNodes.Contains(e.Node))
    {
        backgroundColor = Color.Yellow;
    }
    else if (contentHighlightedNodes.Contains(e.Node))
    {
        backgroundColor = Color.Orange;
    }

    if (e.Node.Tag != null)
    {
        string path = e.Node.Tag.ToString();
        if (File.Exists(path))
        {
            string ext = Path.GetExtension(path);
            if (fileTypeColorMap.ContainsKey(ext))
            {
                textColor = fileTypeColorMap[ext];
            }
            else
            {
                textColor = fileTypeColorMap["default"];
            }
        }
        else if (Directory.Exists(path))
        {
            textColor = Color.FromArgb(52, 77, 221);
        }
        else if (e.Node.Text == "【权限不足，无法访问】")
        {
            textColor = Color.Gray;
        }
    }

    if ((e.State & TreeNodeStates.Selected) != 0)
    {
        e.Graphics.FillRectangle(SystemBrushes.Highlight, e.Bounds);
        TextRenderer.DrawText(
            e.Graphics,
            e.Node.Text,
            nodeFont,
            e.Bounds,
            SystemColors.HighlightText,
            TextFormatFlags.GlyphOverhangPadding
        );
    }
    else
    {
        e.Graphics.FillRectangle(new SolidBrush(backgroundColor), e.Bounds);
        TextRenderer.DrawText(
            e.Graphics,
            e.Node.Text,
            nodeFont,
            e.Bounds,
            textColor,
            TextFormatFlags.GlyphOverhangPadding
        );
    }

    e.DrawDefault = false;
}

private void TreeViewDirectory_NodeMouseDoubleClick(object sender, TreeNodeMouseClickEventArgs e)
{
    if (e.Node.Tag == null) return;
    string path = e.Node.Tag.ToString();

    if (File.Exists(path))
    {
        try
        {
            string ext = Path.GetExtension(path).ToLower();
            // 统一封装路径，添加双引号处理特殊字符
            string quotedPath = string.Format("\"{0}\"", path);

            switch (ext)
            {
                case ".txt":
                    System.Diagnostics.Process.Start("notepad.exe", quotedPath);
                    break;
                case ".md":
                    System.Diagnostics.Process.Start("typora.exe", quotedPath);
                    break;
                case ".xls":
                case ".xlsx":
                    System.Diagnostics.Process.Start("wps.exe", quotedPath);
                    break;
                default:
                    // 对于未知类型，使用系统默认程序打开（推荐用ProcessStartInfo方式更稳定）
                    ProcessStartInfo psi = new ProcessStartInfo
                    {
                        FileName = path,
                        UseShellExecute = true
                    };
                    Process.Start(psi);
                    break;
            }
        }
        catch (Exception ex)
        {
            MessageBox.Show(string.Format("打开文件失败：{0}", ex.Message), "错误", MessageBoxButtons.OK, MessageBoxIcon.Error);
        }
    }
}

private void SelectFileInTreeView(string filePath)
{

    {

        treeViewDirectory.SelectedNode.EnsureVisible();
    }
}

private TreeNode FindNodeByPath(TreeNodeCollection nodes, string path)
{
    foreach (TreeNode node in nodes)
    {
        if (node.Tag != null && node.Tag.ToString() == path)
        {
            return node;
        }

        TreeNode foundNode = FindNodeByPath(node.Nodes, path);
        if (foundNode != null)
        {
            node.Expand();
            return foundNode;
        }
    }
    return null;
}

private void MenuRename_Click(object sender, EventArgs e)
{
    TreeNode selectedNode = treeViewDirectory.SelectedNode;
    if (selectedNode == null || selectedNode.Tag == null) return;

    string path = selectedNode.Tag.ToString();
    string newName = PromptForNewName(selectedNode.Text);

    if (string.IsNullOrEmpty(newName) || newName == selectedNode.Text)
    {
        return;
    }

    string newPath = Path.Combine(Path.GetDirectoryName(path), newName);
    string parentPath = Path.GetDirectoryName(path);

    try
    {
        if (Directory.Exists(path))
        {
            Directory.Move(path, newPath);
        }
        else if (File.Exists(path))
        {
            File.Move(path, newPath);
        }
        // 刷新后定位到父级
        RefreshTreeView(parentPath);
        SelectFileInTreeView(newPath);
        ShowNotification("重命名成功！");
    }
    catch (Exception ex)
    {
        ShowNotification(string.Format("重命名失败：{0}", ex.Message), Color.Red);
    }
}

private void MenuDelete_Click(object sender, EventArgs e)
{
    TreeNode selectedNode = treeViewDirectory.SelectedNode;
    if (selectedNode == null || selectedNode.Tag == null) return;

    string path = selectedNode.Tag.ToString();
    // 记录父级路径（如果是根目录，父级就是自身）
    string parentPath = Path.GetDirectoryName(path);
    // 处理根目录的情况：如果删除的是根目录下的直接子项，父级是根目录
    if (string.IsNullOrEmpty(parentPath) || parentPath == rootFolderPath)
    {
        parentPath = rootFolderPath;
    }

    DialogResult result = MessageBox.Show(string.Format("确定要删除：{0}？", path), "确认删除", MessageBoxButtons.YesNo, MessageBoxIcon.Warning);
    if (result != DialogResult.Yes) return;

    try
    {
        if (Directory.Exists(path))
        {
            Directory.Delete(path, true);
        }
        else if (File.Exists(path))
        {
            File.Delete(path);
        }
        // 刷新后定位到父级目录
        RefreshTreeView(parentPath);
        SelectFileInTreeView(parentPath);
        ShowNotification("删除成功！");
    }
    catch (Exception ex)
    {
        ShowNotification(string.Format("删除失败：{0}", ex.Message), Color.Red);
    }
}

private void LoadDirectoryToTreeView(string path, TreeNodeCollection parentNodes)
{
    try
    {
        TreeNode folderNode = new TreeNode(Path.GetFileName(path));
        folderNode.Tag = path;
        folderNode.ImageIndex = 0;
        parentNodes.Add(folderNode);

        string[] subFolders = Directory.GetDirectories(path);
        foreach (string subFolder in subFolders)
        {
            LoadDirectoryToTreeView(subFolder, folderNode.Nodes);
        }

        string[] files = Directory.GetFiles(path);
        foreach (string file in files)
        {
            TreeNode fileNode = new TreeNode(Path.GetFileName(file));
            fileNode.Tag = file;
            fileNode.ImageIndex = 1;
            folderNode.Nodes.Add(fileNode);
        }
    }
    catch (UnauthorizedAccessException)
    {
        TreeNode errorNode = new TreeNode("【权限不足，无法访问】");
        parentNodes.Add(errorNode);
    }
    catch (Exception ex)
    {
        MessageBox.Show(string.Format("加载目录失败：{0}", ex.Message), "错误", MessageBoxButtons.OK, MessageBoxIcon.Error);
    }
}

// 重载1：默认刷新（定位到根目录）
private void RefreshTreeView()
{
    RefreshTreeView(rootFolderPath);
}

// 重载2：刷新后定位到指定路径
private void RefreshTreeView(string targetPath)
{
    if (!string.IsNullOrEmpty(rootFolderPath))
    {
        treeViewDirectory.Nodes.Clear();
        LoadDirectoryToTreeView(rootFolderPath, treeViewDirectory.Nodes);
        // 找到并展开目标路径的节点
        TreeNode targetNode = FindNodeByPath(treeViewDirectory.Nodes, targetPath);
        if (targetNode != null)
        {
            targetNode.Expand();
            treeViewDirectory.SelectedNode = targetNode;
            targetNode.EnsureVisible();
        }
        else
        {
            // 如果目标路径不存在，展开根节点
            treeViewDirectory.Nodes[0].Expand();
        }
    }
}

private void BtnSearch_Click(object sender, EventArgs e)

    if (btnSearch.Text == "搜索")
    {
        string searchTerm = PromptForSearchTerm();
        if (!string.IsNullOrEmpty(searchTerm))
        {
            SearchAndHighlightNodes(treeViewDirectory.Nodes, searchTerm);
            btnSearch.Text = "取消";
        }
    }
    else
    {
        ClearHighlight();
        btnSearch.Text = "搜索";
    }
}

// 执行搜索
private void SearchAndHighlightNodes(TreeNodeCollection nodes, string searchTerm)
{
    foreach (TreeNode node in nodes)
    {
        bool isTitleMatch = false;
        bool isContentMatch = false;

        // 检查节点文本是否包含搜索关键字
        if (node.Text.IndexOf(searchTerm, StringComparison.OrdinalIgnoreCase) >= 0)
        {
            isTitleMatch = true;
            highlightedNodes.Add(node);
        }
        else if (node.Tag != null && File.Exists(node.Tag.ToString()))
        {
            // 如果是文件，读取文件内容并检查是否包含搜索关键字
            string filePath = node.Tag.ToString();
            string ext = Path.GetExtension(filePath).ToLower();

            // 仅对文本文件进行内容搜索
            if (ext == ".txt" || ext == ".md" || ext == ".cs" || ext == ".java" || ext == ".py" || ext == ".html" || ext == ".js" || ext == ".css")
            {
                try
                {
                    string content = File.ReadAllText(filePath);
                    if (content.IndexOf(searchTerm, StringComparison.OrdinalIgnoreCase) >= 0)
                    {
                        isContentMatch = true;
                        contentHighlightedNodes.Add(node);
                    }
                }
                catch (Exception ex)
                {
                    MessageBox.Show(string.Format("读取文件失败：{0}", ex.Message), "错误", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
        }

        if (isTitleMatch || isContentMatch)
        {
            node.EnsureVisible();
        }

        // 递归搜索子节点
        SearchAndHighlightNodes(node.Nodes, searchTerm);
    }

    treeViewDirectory.Invalidate();
}

// 时间戳转日期按钮事件
private void BtnTsToDate_Click(object sender, EventArgs e)
{
    if (string.IsNullOrEmpty(txtTsInput.Text) || !long.TryParse(txtTsInput.Text, out long ts))
    {
        ShowStatus("请输入有效的时间戳", Color.Orange);
        return;
    }

    string unit = cboTsUnit.SelectedItem.ToString();
    long timestamp = unit == "毫秒" ? ts : ts * 1000;

    try
    {
        DateTime date = new DateTime(1970, 1, 1, 0, 0, 0, DateTimeKind.Utc).AddMilliseconds(timestamp).ToLocalTime();
        txtTsResult.Text = date.ToString("yyyy-MM-dd HH:mm:ss");
    }
    catch
    {
        ShowStatus("时间戳超出范围", Color.Red);
    }
}

// 日期转时间戳按钮事件
private void BtnDateToTs_Click(object sender, EventArgs e)
{
    if (string.IsNullOrEmpty(txtDateInput.Text))
    {
        ShowStatus("请输入日期", Color.Orange);
        return;
    }

    DateTime date;
    // 尝试多种格式解析
    string[] formats = { "yyyy-MM-dd HH:mm:ss", "yyyy/MM/dd HH:mm:ss", "yyyy-MM-dd", "yyyy/MM/dd" };
    if (!DateTime.TryParseExact(txtDateInput.Text, formats, CultureInfo.InvariantCulture, DateTimeStyles.None, out date))
    {
        ShowStatus("日期格式不正确", Color.Red);
        return;
    }

    // 转换为时间戳（1970-01-01 00:00:00 UTC）
    DateTime epoch = new DateTime(1970, 1, 1, 0, 0, 0, DateTimeKind.Utc);
    long ts = (long)(date.ToUniversalTime() - epoch).TotalMilliseconds;

    string unit = cboDateUnit.SelectedItem.ToString();
    long result = unit == "毫秒" ? ts : ts / 1000;
    txtDateResult.Text = result.ToString();
}
