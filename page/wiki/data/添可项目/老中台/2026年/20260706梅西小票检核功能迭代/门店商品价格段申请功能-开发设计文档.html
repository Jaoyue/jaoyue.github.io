2.2 索引设计说明

索引 用途
idx_store_segment_code 申请单号模糊查询 + 精确匹配
idx_audit_status 按审批状态筛选申请单
idx_creation_date 按创建时间范围查询 + 排序
idx_product_line_id 核心查询：按商品行ID查找关联门店（审批后生成明细用）
idx_product_store 明细表通用组合查询：产品 + 门店

@Override

@Transactional(rollbackFor = Exception.class)
public ImsProductPriceStoreSegmentHeader saveOrUpdate(
        ImsProductPriceStoreSegmentHeaderBo bo) throws Exception {

    ImsProductPriceStoreSegmentHeader header = new ImsProductPriceStoreSegmentHeader();
    BeanUtils.copyProperties(bo, header);

    // 1. 设置审批状态
    if (Boolean.TRUE.equals(bo.getSaveonly())) {
        header.setAuditStatus("DRAFT");   // 仅保存
    } else {
        header.setAuditStatus("APPROVAL"); // 提交审批
        validateSubmit(header);             // 前置校验
    }

    // 2. 新增时生成申请单号
    if (header.getStoreSegmentHeaderId() == null) {
        header.setStoreSegmentCode(
            saafSharedSerialApiClient.getSegmentId(
                ImsProductConstants.STORE_PRICE_SEGMENT_BIZ_KEY));
    }

    // 3. 保存（MyBatis-Plus 自动判断 insert/update）
    this.saveOrUpdate(header);

    // 4. 提交流程到BPM
    bo.setStoreSegmentHeaderId(header.getStoreSegmentHeaderId());
    JSONObject businessData = JSON.parseObject(JSON.toJSONString(header));
    ImsProductPriceStoreSegmentHeaderVo vo = findById(header.getStoreSegmentHeaderId());
    businessData.put("createdBy", vo.getCreatedBy());
    doCommit(bo, businessData);

    return header;
}

@Override
@Transactional(rollbackFor = Exception.class)
public ImsProductPriceStoreSegmentHeader saveOrUpdateAll(
        ImsProductPriceStoreSegmentHeaderRelationBo bo) throws Exception {

    // 1. 保存头表
    ImsProductPriceStoreSegmentHeaderBo headerBo =
        bo.getImsProductPriceStoreSegmentHeaderBo();
    Assert.notNull(headerBo, LanUtil.getMessage(bo.getLan(),
        "cannotBeEmpty", new Object[]{"imsProductPriceStoreSegmentHeaderBo"}));

    ImsProductPriceStoreSegmentHeader header = saveOrUpdate(headerBo);
    if (header == null) return null;

    // 2. 更新场景：先删除旧的行表数据
    boolean isUpdate = headerBo.getStoreSegmentHeaderId() != null;
    if (isUpdate) {
        // 删除旧商品行（逻辑删除）
        productLineService.deleteByHeaderId(header.getStoreSegmentHeaderId());
        // 删除旧门店行（逻辑删除）
        storeLineService.deleteByHeaderId(header.getStoreSegmentHeaderId());
    }

    // 3. 逐个保存商品行 → 保存该商品的门店行
    List<ImsProductPriceStoreSegmentProductLineBo> productBoList =
        bo.getImsProductPriceStoreSegmentProductLine();
    if (!CollectionUtils.isEmpty(productBoList)) {
        for (ImsProductPriceStoreSegmentProductLineBo productBo : productBoList) {
            productBo.setStoreSegmentHeaderId(header.getStoreSegmentHeaderId());
            
            // 3.1 保存商品行
            ImsProductPriceStoreSegmentProductLine productLine = new ImsProductPriceStoreSegmentProductLine();
            BeanUtils.copyProperties(productBo, productLine);
            productLineService.saveOrUpdate(productLine);
            
            // 3.2 保存该商品关联的门店行
            List<ImsProductPriceStoreSegmentStoreLineBo> storeBoList = productBo.getStoreLines();
            if (!CollectionUtils.isEmpty(storeBoList)) {
                for (ImsProductPriceStoreSegmentStoreLineBo storeBo : storeBoList) {
                    storeBo.setStoreSegmentHeaderId(header.getStoreSegmentHeaderId());
                    storeBo.setStoreSegmentProductLineId(productLine.getStoreSegmentProductLineId());
                    
                    ImsProductPriceStoreSegmentStoreLine storeLine = new ImsProductPriceStoreSegmentStoreLine();
                    BeanUtils.copyProperties(storeBo, storeLine);
                    storeLineService.saveOrUpdate(storeLine);
                }
            }
        }
    }

    return header;
}

            throw new BaseRuntimeException(message);
        } else {
            throw new BaseRuntimeException("调用审批流失败！");
        }
    }
}

8.2.7 validateSubmit 提交校验

private void validateSubmit(ImsProductPriceStoreSegmentHeader header) {
    // 1. 检查商品行
    List<ImsProductPriceStoreSegmentProductLine> productList =
        productLineService.findByProperty("store_segment_header_id",
            header.getStoreSegmentHeaderId());
    if (productList.isEmpty()) {
        throw new BaseRuntimeException("存在商品行为空，不允许提交");
    }

    // 2. 检查门店行
    List<ImsProductPriceStoreSegmentStoreLine> storeList =
        storeLineService.findByProperty("store_segment_header_id",
            header.getStoreSegmentHeaderId());
    if (storeList.isEmpty()) {
        throw new BaseRuntimeException("存在门店行为空，不允许提交");
    }

    // 3. 校验价格合理性：price_min <= price_max
    for (ImsProductPriceStoreSegmentProductLine line : productList) {
        if (line.getPriceMin() != null && line.getPriceMax() != null
                && line.getPriceMin().compareTo(line.getPriceMax()) > 0) {
            throw new BaseRuntimeException(
                "产品[" + line.getProductSapId() + "]价格下限不得大于价格上限");
        }
    }

    // 4. 时间段唯一性校验：同一产品+门店在重叠时间段不能有审批中单据
    QueryImsProductPriceStoreSegmentHeader query = new QueryImsProductPriceStoreSegmentHeader();
    query.setStartDate(header.getStartDate());
    query.setEndDate(header.getEndDate());
    query.setProductSapIdList(productList.stream()
        .map(ImsProductPriceStoreSegmentProductLine::getProductSapId)
        .collect(Collectors.toList()));
    query.setStoreCodeList(storeList.stream()
        .map(ImsProductPriceStoreSegmentStoreLine::getStoreCode)
        .collect(Collectors.toList()));

    List<ImsProductPriceStoreSegmentHeaderVo> conflictList = mapper.findBySubmit(query);
    if (!conflictList.isEmpty()) {
        throw new BaseRuntimeException("存在相同时间范围内的审批中单据，不允许提交");
    }
}

@Override
@Transactional(rollbackFor = Exception.class)
public ImsProductPriceStoreSegmentHeader updateStatusById(
        @TransMsgParam Long messageIndex,
        ActBpmProcessCallbackBo actBpmProcessCallbackBo) {

    // 1. 查询申请单
    ImsProductPriceStoreSegmentHeaderVo vo =
        this.findById(Long.valueOf(actBpmProcessCallbackBo.getId()));
    ImsProductPriceStoreSegmentHeader header = new ImsProductPriceStoreSegmentHeader();
    BeanUtils.copyProperties(vo, header);

    // 2. 状态幂等：已是指定状态则跳过
    if (actBpmProcessCallbackBo.getStatus().equals(header.getAuditStatus())) {
        return header;
    }

    // 3. 审批通过 → 生成价格段明细
    if ("ALLOW".equals(actBpmProcessCallbackBo.getStatus())) {
        generateSegmentPrices(header);
    }

    // 4. 更新审批状态
    header.setAuditStatus(actBpmProcessCallbackBo.getStatus());
    mapper.updateById(header);
    return header;
}

/**
 * 审批通过后，按商品×门店实际关联展开生成价格段明细
 */
private void generateSegmentPrices(ImsProductPriceStoreSegmentHeader header) {
    // 查询所有商品行
    List<ImsProductPriceStoreSegmentProductLine> productLines =
        productLineService.findByProperty("store_segment_header_id",
            header.getStoreSegmentHeaderId());

    List<ImsProductPriceStoreSegment> segmentList = new ArrayList<>();

    for (ImsProductPriceStoreSegmentProductLine product : productLines) {
        // 查询该商品关联的门店行
        QueryWrapper<ImsProductPriceStoreSegmentStoreLine> storeWrapper =
            new QueryWrapper<ImsProductPriceStoreSegmentStoreLine>()
                .eq("store_segment_product_line_id", product.getStoreSegmentProductLineId())
                .eq("delete_flag", 0);
        List<ImsProductPriceStoreSegmentStoreLine> storeLines =
            storeLineMapper.selectList(storeWrapper);

        for (ImsProductPriceStoreSegmentStoreLine store : storeLines) {
            ImsProductPriceStoreSegment segment = new ImsProductPriceStoreSegment();
            // 产品信息
            segment.setProductSapId(product.getProductSapId());
            segment.setProductName(product.getProductName());
            segment.setProductBigType(product.getProductBigType());
            segment.setProductMiddleType(product.getProductMiddleType());
            segment.setProductSmallType(product.getProductSmallType());
            segment.setPriceMin(product.getPriceMin());
            segment.setPriceMax(product.getPriceMax());
            // 门店信息
            segment.setStoreCode(store.getStoreCode());
            segment.setStoreName(store.getStoreName());
            segment.setRetailType(store.getRetailType());
            segment.setCustomerCode(store.getCustomerCode());
            segment.setCustomerName(store.getCustomerName());
            // 时间和关联信息
            segment.setStartDate(header.getStartDate());
            segment.setEndDate(header.getEndDate());
            segment.setSegmentCode(header.getStoreSegmentCode());
            segment.setEffectiveStatus(ImsProductConstants.EFFECTIVE);

            segmentList.add(segment);
        }
    }

    if (!segmentList.isEmpty()) {
        segmentService.saveOrUpdateAll(segmentList);
    }
}

        outputStream.flush();
    } catch (IOException e) {
        log.error("门店商品价格段导出异常 {} -- {}", query.getStoreSegmentHeaderId(), e);
    }
}

关键注意事项

10.1 审批流配置

BPM 流程定义 Key：需要在 BPM 管理平台创建一个新的流程定义，Key 建议为 "STORE_PRICE_SEGMENT"（由前端传入 processDefinitionKey 字段）。

回调地址：POST /imsProductPriceStoreSegmentHeader/updateStatus（ActBpmProcessCallbackBo 会自动携带审批结果）。

BPM 必须字段（在 HeaderBo 和 DeleteBo 中带 @NotNull）：

字段 来源 说明
processDefinitionKey 前端传入 BPM流程定义Key
orgId 前端传入 所属OU组织
responsibilityId 前端传入 职责ID
departmentId 前端传入 部门ID（可选）
positionId 前端传入 职位ID（可选）

10.2 BPM 回调接口的安全注意

updateStatus 接口仅接受 BPM 引擎的回调，应不加用户认证拦截（参照现有实现）。Controller 中不调用 super.validatorData(result)，直接接受 ActBpmProcessCallbackBo。

10.3 事务与锁

方法 事务注解 说明
saveOrUpdate @GlobalTransactional + @Transactional 分布式事务：BPM调用可能跨服务
saveOrUpdateAll @Transactional 仅本地事务，不包含BPM调用
updateStatusById @Transactional 本地事务：更新头表+生成明细
deleteAll @Transactional 本地事务：有BPM调用但仅在非DRAFT时触发
batchDelete @Transactional 简单批量删除

注意：saveOrUpdateAll 调用 saveOrUpdate（后者已有 @GlobalTransactional），所以 saveOrUpdateAll 自己的 @Transactional 会参与到外层 @GlobalTransactional 中，不会重复。

10.4 流水号编码规则配置

需要在流水号管理平台配置 BizKey："IMSSTOREPRICESEGMENT"，格式建议：PSA-yyyyMMdd-序号（如 PSA-20260708-0001）。

10.5 审批状态码值字典

使用已有字典 "imsProductAuditStatus"（与促销价目表一致），建议在数据字典中新增状态码：

  • DRAFT — 草稿
  • APPROVAL — 待审批
  • ALLOW — 已审批
  • REJECT — 已驳回

10.6 数据聚合/转义

VO字段 注解 说明
auditStatus @SieEscapeField(BASE_LOOK_UP, "imsProductAuditStatus") 审批状态码值转义
createdBy @SieEscapeField(BASE_USER_ID) 用户ID→用户名
storeCode @SieDataAggregate(MDM_BASE_CUSTOMER) 门店主数据聚合（门店名称、零售业态等）
productSapId @SieDataAggregate(MDM_BASE_PRODUCT) 产品主数据聚合（产品名称、大类等）

10.7 前端联调注意

  1. saveAll 接口的数据结构：商品行内嵌门店行 storeLines，后端按此嵌套结构平铺保存
  2. 审批状态不可编辑：APPROVAL 和 ALLOW 状态的申请单，前端应禁用编辑和删除按钮
  3. SAP ID 自动带出：前端输入 SAP ID 后失焦调用产品查询接口自动带出名称、大类等信息（后端也做校验）
  4. 门店编码自动带出：同理，调用主数据查询接口自动带出

10.8 性能注意事项

  1. 分页大小：默认 ImsProductConstants.PAGE_SIZE = 1000，但导出时应放开限制
  2. 笛卡尔积规模：如果商品×门店组合过多（如 100商品 × 500门店 = 50000条），审批回调的 generateSegmentPrices 会批量插入大量数据，建议使用 saveBatch 分批处理
  3. 索引覆盖：明细表 idx_date_range 和 idx_product_store 组合索引确保查询性能
